### IV. SYNTHETIC VERIFICATION SCRIPT (TARGET SWITCHING & SUB-MENU PARSING)

Execute this script via vitest or Node to verify dynamic target resolution and
menu parsing in memory:

```typescript
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
import { fetchStorageRecords } from './packages/132.github/03.storage.unit/buz/storage.buzz.js'
import { auditHashChainIntegrity } from './packages/132.github/00.github.unit/buz/github.buzz.js'
import { StorageModel } from './packages/132.github/03.storage.unit/storage.model.js'
import { GithubModel } from './packages/132.github/00.github.unit/github.model.js'

describe('Synthetic Target Switching & Sub-Menu Verification', () => {
  beforeEach(() => {
    vi.restoreAllMocks()
    ;(global as any).LIBRARY = { hunt: vi.fn().mockResolvedValue({}) }
  })

  afterEach(() => {
    delete (global as any).agentBaseUrl
    delete (global as any).githubBaseUrl
  })

  it('redirects storage.unit queries to local worker port 8787 when agentBaseUrl is set', async () => {
    const model = new StorageModel()
    const slv = vi.fn()
    let interceptedUrl = ''

    ;(global as any).agentBaseUrl =
      '[http://127.0.0.1:8787](http://127.0.0.1:8787)'

    global.fetch = vi.fn().mockImplementation(async (url: string) => {
      interceptedUrl = url
      return new Response(JSON.stringify({ ok: true, events: [] }), {
        status: 200,
      })
    }) as any

    await fetchStorageRecords(model, { idx: 'fetch', val: 50, slv }, {} as any)
    expect(slv).toHaveBeenCalled()
    expect(interceptedUrl).toContain(
      '[http://127.0.0.1:8787/api/audit/recent?limit=50](http://127.0.0.1:8787/api/audit/recent?limit=50)',
    )
  })

  it('redirects github.unit chain audit queries to local worker port 8787 when agentBaseUrl is set', async () => {
    const model = new GithubModel()
    const slv = vi.fn()
    let interceptedUrl = ''

    ;(global as any).agentBaseUrl =
      '[http://127.0.0.1:8787](http://127.0.0.1:8787)'

    global.fetch = vi.fn().mockImplementation(async (url: string) => {
      interceptedUrl = url
      return new Response(JSON.stringify({ ok: true, count: 0, rows: [] }), {
        status: 200,
      })
    }) as any

    await auditHashChainIntegrity(
      model,
      { idx: 'audit', src: 'astro-kahn-it-com/001.goblin-lore', slv },
      {} as any,
    )
    expect(slv).toHaveBeenCalled()
    expect(interceptedUrl).toContain(
      '[http://127.0.0.1:8787/api/audit/verify-chain](http://127.0.0.1:8787/api/audit/verify-chain)',
    )
  })

  it('falls back cleanly to live worker URL when agentBaseUrl is absent', async () => {
    const model = new StorageModel()
    const slv = vi.fn()
    let interceptedUrl = ''

    delete (global as any).agentBaseUrl

    global.fetch = vi.fn().mockImplementation(async (url: string) => {
      interceptedUrl = url
      return new Response(JSON.stringify({ ok: true, events: [] }), {
        status: 200,
      })
    }) as any

    await fetchStorageRecords(model, { idx: 'fetch', val: 10, slv }, {} as any)
    expect(slv).toHaveBeenCalled()
    expect(interceptedUrl).toContain('repo-bot-00.berad4000.workers.dev')
  })
})
```
