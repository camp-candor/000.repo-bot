### II. ARCHITECTURAL MISSION & REQUIREMENTS

When a Pull Request is merged manually via GitHub's Web UI (e.g., clicking the
green **[Merge pull request]** button on `github.com`), the outbound Repo-Bot
merge executor is bypassed[cite: 7].

Implement inbound GitHub Webhook ingestion in `apps/worker` to catch manual
GitHub UI merges and broadcast expressive, explicit notifications to
`#ops-bridge` (`SLACK_CHANNEL_ID`):

1. **Origin Discrimination (Repo-Bot Engine vs. GitHub Web UI):**
   - Inspect incoming `pull_request.closed` webhook events where
     `payload.pull_request.merged === true`[cite: 7].
   - Distinguish between merges initiated by Repo-Bot's Compare-And-Swap (CAS)
     executor and manual UI merges:
     - Repo-Bot merges contain commit message markers (`Audited-Head-SHA:` or
       `squash merge completed by repo-bot`) or are initiated by the bot user.
     - Any merge lacking these markers or initiated by an interactive user
       (e.g., `payload.pull_request.merged_by.login`) must be identified as:
       `origin: 'GITHUB_MANUAL_UI'`.

2. **Expressive Slack Announcement (`apps/worker/src/slackBridge.ts`):**
   - Extend `postSlackMergeAnnouncement` (or parameterize
     `MergeAnnouncementParams`) to support
     `origin: 'REPO_BOT_CAS' | 'GITHUB_MANUAL_UI'`.
   - For `GITHUB_MANUAL_UI` merges, the Block Kit card must explicitly announce:
     - Header: `:: GITHUB MANUAL UI MERGE DETECTED (PR #${pullNumber})`
     - Callout Banner:
       `[ORIGIN: DIRECT GITHUB WEB CONSOLE / MANUAL HUMAN ACTION]`
     - Advisory Notice:
       `Advisory: Merged directly on GitHub.com outside the Repo-Bot CAS verification pipeline.`
     - Metadata Fields:
       - **Repository:** `owner/repo`
       - **PR Link:** `<url|#pullNumber Title>`
       - **Merged By:** GitHub user login (`@login`)
       - **Target Trunk:** Base branch (e.g., `main`)
       - **Merge Commit SHA:** Full / truncated merge SHA
       - **Head Branch:** Head ref name (e.g., `feat-blessed-merge-...`)

3. **Durable Object Self-Healing Reconcile (`RepoBotDO`):**
   - If a matching `taskId` or branch ref can be identified for the manual
     merge, post `MERGE_SUCCEEDED` to the DO (`/fsm/transition`) so in-flight
     state machines do not remain stranded in `AWAITING_APPROVAL` or trigger
     false-positive 24h watchdog expiration alarms.

4. **Dedicated Unit Test Coverage (`githubWebhookManualMerge.test.ts`):**
   - Verify HMAC signature verification of incoming `pull_request.closed`
     webhooks.
   - Verify manual UI merges are correctly identified as `GITHUB_MANUAL_UI`.

/**
 * Verifies Slack incoming webhook HMAC-SHA256 signature.
 */
export async function verifySlackSignature(
  rawBody: string,
  headers: { timestamp?: string | null; signature?: string | null },
  signingSecret: string,
): Promise<VerificationResult> {
  const { timestamp, signature } = headers

  if (!timestamp || !signature) {
    return { valid: false, reason: 'MISSING_SLACK_SIGNATURE_HEADERS' }
  }

  const fiveMinutesAgo = Math.floor(Date.now() / 1000) - 60 * 5
  const reqTimestamp = parseInt(timestamp, 10)
  if (isNaN(reqTimestamp) || reqTimestamp < fiveMinutesAgo) {
    return { valid: false, reason: 'SLACK_REQUEST_TIMESTAMP_EXPIRED' }
  }

  const sigBasestring = `v0:${timestamp}:${rawBody}`
  const encoder = new TextEncoder()

  const cryptoKey = await crypto.subtle.importKey(
    'raw',
    encoder.encode(signingSecret),
    { name: 'HMAC', hash: 'SHA-256' },
    false,
    ['sign'],
  )

  const signatureBuffer = await crypto.subtle.sign(
    'HMAC',
    cryptoKey,
    encoder.encode(sigBasestring),
  )

  const hashHex = Array.from(new Uint8Array(signatureBuffer))
    .map((b) => b.toString(16).padStart(2, '0'))
    .join('')
  const expectedSignature = `v0=${hashHex}`

  if (signature.length !== expectedSignature.length) {
    return { valid: false, reason: 'SIGNATURE_LENGTH_MISMATCH' }
  }

  let mismatch = 0
  for (let i = 0; i < signature.length; i++) {
    mismatch |= signature.charCodeAt(i) ^ expectedSignature.charCodeAt(i)
  }

  return mismatch === 0
    ? { valid: true }
    : { valid: false, reason: 'HMAC_VERIFICATION_FAILED' }
}

  const shortHeadSha = (params.auditedHeadSha || '0000000').slice(0, 7)

const handleGitHubWebhook = async (c: any) => {
  const rawBody = await c.req.text()
  const signature = c.req.header('x-hub-signature-256')
  const githubEvent = c.req.header('x-github-event')

  // 1. Signature Verification (using normalized GH_WEBHOOK_SECRET)
  const webhookSecret = c.env.GH_WEBHOOK_SECRET || c.env.GITHUB_WEBHOOK_SECRET
  if (webhookSecret) {
    const isValid = await verifyGitHubSignature(
      rawBody,
      signature,
      webhookSecret,
    )
    if (!isValid) {
      console.error(
        '>> [WEBHOOK REJECTED] Invalid GitHub HMAC-SHA256 signature',
      )
      return c.json({ error: 'UNAUTHORIZED_SIGNATURE' }, 401)
    }
  }

  let payload: any = {}
  try {
    payload = JSON.parse(rawBody)
  } catch {
    return c.text('Malformed JSON payload', 400)
  }

  // 2. Process Pull Request Closed & Merged Event
  if (
    githubEvent === 'pull_request' &&
    payload.action === 'closed' &&
    payload.pull_request?.merged === true
  ) {
    const pr = payload.pull_request
    const repo = payload.repository?.name || '000.repo-bot'
    const owner = payload.repository?.owner?.login || 'camp-candor'
    const pullNumber = pr.number
    const mergeCommitSha = pr.merge_commit_sha || ''
    const headSha = pr.head?.sha || ''
    const baseRef = pr.base?.ref || 'main'
    const headRef = pr.head?.ref || ''
    const mergedBy = pr.merged_by?.login || payload.sender?.login || 'unknown'
    const prTitle = pr.title || ''

    // Distinguish between Repo-Bot CAS squash merge and Direct Manual GitHub UI merge
    const commitMessage = pr.body || ''
    const isRepoBotCAS =
      commitMessage.includes('squash merge completed by repo-bot') ||
      commitMessage.includes('Audited-Head-SHA:') ||
      mergedBy.includes('repo-bot')

    const origin = isRepoBotCAS ? 'REPO_BOT_CAS' : 'GITHUB_MANUAL_UI'
    console.log(
      `>> [GITHUB WEBHOOK] Ingested PR #${pullNumber} closed & merged. Origin: ${origin} (Merged by: ${mergedBy})`,
    )

    // Extract taskId from headRef if following spec/{taskId}-{sha} convention
    const specMatch = headRef.match(
      /^spec\/([a-zA-Z0-9._-]+?)(-[a-f0-9]{7,40})?$/,
    )
    const taskId = specMatch ? specMatch[1] : `PR-${pullNumber}`

    // A. Broadcast announcement to #ops-bridge
    c.executionCtx.waitUntil(
      postSlackMergeAnnouncement(
        {
          taskId,
          pullNumber,
          mergeCommitSha,
          auditedHeadSha: headSha,
          targetBranch: baseRef,
          branchName: headRef,
          actor: mergedBy,
          owner,
          repo,
          prTitle,
          origin,
        },
        c.env,
      ),
    )

    // B. Reconcile with RepoBotDO to prevent stranded FSM states
    if (c.env.REPO_BOT_DO) {
      c.executionCtx.waitUntil(
        (async () => {
          try {
            const doId = c.env.REPO_BOT_DO.idFromName(taskId)
            const taskDO = c.env.REPO_BOT_DO.get(doId)
            await taskDO.fetch(
              new Request('https://internal/fsm/transition', {
                method: 'POST',

                  taskId,
                  headSha,
                  mergeCommitSha,

            console.warn(

              doErr.message,
            )
          }
        })(),
      )
    }

    return c.json({ ok: true, status: 'MERGE_RECORDED', origin, pullNumber })
  }

  return c.json({ ok: true, status: 'EVENT_RECEIVED', event: githubEvent })
}
