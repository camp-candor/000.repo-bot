```typescript
/**
 * apps/worker/src/fencing/tokenSequencer.ts
 *
 * Edge-native monotonic epoch sequencer and concurrency fencing manager.
 * Enforces gapless N -> N+1 progression and rejects stale zombie mutations.
 * Pure 7-bit clean ASCII compliance.
 */

export interface TaskLease {
  readonly taskId: string
  readonly epochToken: number
  readonly leasedAt: number
  readonly expiresAt: number
  readonly owner: string
}

export interface FencingValidationResult {
  readonly valid: boolean
  readonly activeEpoch: number
  readonly submittedEpoch: number
  readonly reason?: string
}

export const DEFAULT_LEASE_TTL_MS = 180000 // 3 minutes

export class MonotonicTokenSequencer {
  private globalEpoch: number = 0
  private readonly leases: Map<string, TaskLease> = new Map()

  constructor(initialEpoch: number = 0) {
    if (!Number.isInteger(initialEpoch) || initialEpoch < 0) {
      throw new Error('INITIAL_EPOCH_MUST_BE_NON_NEGATIVE_INTEGER')
    }
    this.globalEpoch = initialEpoch
  }

  /**
   * Retrieves the current highest issued global monotonic epoch.
   */
  public getCurrentEpoch(): number {
    return this.globalEpoch
  }

  /**
   * Advances the global epoch monotonically (N -> N+1).
   * Prohibits integer skipping or regressive resets.
   */
  public advanceGlobalEpoch(): number {
    this.globalEpoch += 1
    return this.globalEpoch
  }

  /**
   * Issues an authoritative, monotonically fenced lease for a task.
   */
  public acquireTaskLease(
    taskId: string,
    owner: string,
    now: number = Date.now(),
    ttlMs: number = DEFAULT_LEASE_TTL_MS
  ): TaskLease {
    if (!taskId || typeof taskId !== 'string') {
      throw new Error('INVALID_TASK_ID')
    }

    const nextEpoch = this.advanceGlobalEpoch()
    const lease: TaskLease = {
      taskId,
      epochToken: nextEpoch,
      leasedAt: now,
      expiresAt: now + ttlMs,
      owner,
    }

    this.leases.set(taskId, lease)
    return lease
  }

  /**
   * Retrieves the active lease for a specific task if it exists and has not expired.
   */
  public getTaskLease(taskId: string, now: number = Date.now()): TaskLease | null {
    const lease = this.leases.get(taskId)
    if (!lease) {
      return null
    }

    if (now > lease.expiresAt) {
      this.leases.delete(taskId)
      return null
    }

    return lease
  }

  /**
   * Validates an incoming worker submission against the active epoch fence.
   * Rejects zombie workers presenting stale or mismatched epoch tokens.
   */
  public assertEpochFence(
    taskId: string,
    submittedEpoch: number,
    now: number = Date.now()
  ): FencingValidationResult {
    const activeLease = this.getTaskLease(taskId, now)

    if (!activeLease) {
      return {
        valid: false,
        activeEpoch: this.globalEpoch,
        submittedEpoch,
        reason: 'LEASE_NOT_FOUND_OR_EXPIRED',
      }
    }

    if (!Number.isInteger(submittedEpoch) || submittedEpoch <= 0) {
      return {
        valid: false,
        activeEpoch: activeLease.epochToken,
        submittedEpoch,
        reason: 'MALFORMED_EPOCH_TOKEN',
      }
    }

    if (submittedEpoch < activeLease.epochToken) {
      return {
        valid: false,
        activeEpoch: activeLease.epochToken,
        submittedEpoch,
        reason: 'ZOMBIE_EPOCH_MUTATION_REJECTED',
      }
    }

    if (submittedEpoch > activeLease.epochToken) {
      return {
        valid: false,
        activeEpoch: activeLease.epochToken,
        submittedEpoch,
        reason: 'UNAUTHORIZED_FUTURE_EPOCH_TOKEN',
      }
    }

    return {
      valid: true,
      activeEpoch: activeLease.epochToken,
      submittedEpoch,
    }
  }

  /**
   * Renews an active lease if the submitted epoch matches the active lease token.
   */
  public renewTaskLease(
    taskId: string,
    submittedEpoch: number,
    now: number = Date.now(),
    ttlMs: number = DEFAULT_LEASE_TTL_MS
  ): TaskLease {
    const check = this.assertEpochFence(taskId, submittedEpoch, now)
    if (!check.valid) {
      throw new Error(check.reason || 'LEASE_RENEWAL_FENCING_BREACH')
    }

    const currentLease = this.leases.get(taskId)!
    const renewedLease: TaskLease = {
      ...currentLease,
      expiresAt: now + ttlMs,
    }

    this.leases.set(taskId, renewedLease)
    return renewedLease
  }

  /**
   * Releases an active lease, optionally clearing it from memory.
   */
  public releaseTaskLease(
    taskId: string,
    submittedEpoch: number,
    now: number = Date.now()
  ): boolean {
    const check = this.assertEpochFence(taskId, submittedEpoch, now)
    if (!check.valid) {
      return false
    }

    this.leases.delete(taskId)
    return true
  }

  /**
   * Cleans up expired leases from memory to prevent unbounded table growth.
   */
  public pruneExpiredLeases(now: number = Date.now()): number {
    let prunedCount = 0
    for (const [taskId, lease] of this.leases.entries()) {
      if (now > lease.expiresAt) {
        this.leases.delete(taskId)
        prunedCount += 1
      }
    }
    return prunedCount
  }
}
```
