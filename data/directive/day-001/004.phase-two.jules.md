import type { DiffModel } from '../diff.model.js'
import type DiffBit from '../fce/diff.bit.js'
import type { DiffFileEntry } from '../fce/diff.bit.js'
import type State from '../../99.core/state.js'

const MAX_FILE_BLAST_RADIUS = 50

const PROTECTED_INFRASTRUCTURE_PATTERNS = [
  /^\.github\/workflows\/.+/,
  /^tests\/.+/,
  /^vitest\.config\.ts$/,
  /^apps\/995\.library\/995\.library\/.+/,
  /^package\.json$/,
  /^package-lock\.json$/
]

export const initDiff = (cpy: DiffModel, bal: DiffBit, ste: State) => {
  if (bal.slv) bal.slv({ difBit: { idx: 'init-diff', dat: cpy } })
  return cpy
}

export const evaluateBlastRadius = (
  files: DiffFileEntry[]
): { valid: boolean; reason?: string } => {
  if (files.length > MAX_FILE_BLAST_RADIUS) {
    return {
      valid: false,
      reason: 'BLAST_RADIUS_EXCEEDED_MAX_50_FILES'
    }
  }
  return { valid: true }
}

export const evaluateProtectedPaths = (
  files: DiffFileEntry[]
): { valid: boolean; offendingPath?: string; reason?: string } => {
  for (const entry of files) {
    const rawPath = entry.filename

    if (
      rawPath.includes('../') ||
      rawPath.includes('..\\') ||
      rawPath.includes('%2e%2e') ||
      rawPath.includes('\0')
    ) {
      return {
        valid: false,
        offendingPath: rawPath,
        reason: 'DIRECTORY_TRAVERSAL_DETECTED'
      }
    }

    const normalized = rawPath.replace(/\\/g, '/').replace(/^\/+/, '')

    for (const pattern of PROTECTED_INFRASTRUCTURE_PATTERNS) {
      if (pattern.test(normalized)) {
        return {
          valid: false,
          offendingPath: normalized,
          reason: 'PROTECTED_INFRASTRUCTURE_MUTATION_BREACH'
        }
      }
    }
  }

  return { valid: true }
}

export const evaluateWhitelistConformance = (
  files: DiffFileEntry[],
  whitelist?: string[]
): { valid: boolean; unapprovedFiles?: string[] } => {
  if (!whitelist || whitelist.length === 0) {
    return { valid: true }
  }

  const unapproved: string[] = []

  for (const entry of files) {
    const normalized = entry.filename.replace(/\\/g, '/').replace(/^\/+/, '')
    const allowed = whitelist.some((rule) => {
      const cleanRule = rule.replace(/\\/g, '/').replace(/^\/+/, '')
      if (cleanRule.endsWith('/**')) {
        const prefix = cleanRule.slice(0, -3)
        return normalized.startsWith(prefix)
      }
      if (cleanRule.endsWith('/*')) {
        const prefix = cleanRule.slice(0, -2)
        return normalized.startsWith(prefix) && !normalized.slice(prefix.length + 1).includes('/')
      }
      return normalized === cleanRule
    })

    if (!allowed) {
      unapproved.push(normalized)
    }
  }

  if (unapproved.length > 0) {
    return { valid: false, unapprovedFiles: unapproved }
  }

  return { valid: true }
}

export const auditPullRequestDiff = async (
  cpy: DiffModel,
  bal: DiffBit,
  ste: State
) => {
  cpy.totalAudits += 1
  cpy.lastAuditTimestamp = Date.now()

  const files = bal.files || []

  const radiusResult = evaluateBlastRadius(files)
  if (!radiusResult.valid) {
    cpy.failedAudits += 1
    cpy.lastAuditError = radiusResult.reason!
    if (bal.slv) {
      bal.slv({
        difBit: {
          idx: 'audit-pr-diff-fail',
          val: 0,
          dat: { valid: false, reason: radiusResult.reason }
        }
      })
    }
    return cpy
  }

  const pathResult = evaluateProtectedPaths(files)
  if (!pathResult.valid) {
    cpy.failedAudits += 1
    cpy.lastAuditError = pathResult.reason!
    if (bal.slv) {
      bal.slv({
        difBit: {
          idx: 'audit-pr-diff-fail',
          val: 0,
          dat: {
            valid: false,
            reason: pathResult.reason,
            offendingPath: pathResult.offendingPath
          }
        }
      })
    }
    return cpy
  }

  const whitelistResult = evaluateWhitelistConformance(files, bal.fileWhitelist)
  if (!whitelistResult.valid) {
    cpy.failedAudits += 1
    cpy.lastAuditError = 'SCOPE_VIOLATION_UNAPPROVED_PATH'
    if (bal.slv) {
      bal.slv({
        difBit: {
          idx: 'audit-pr-diff-fail',
          val: 0,
          dat: {
            valid: false,
            reason: 'SCOPE_VIOLATION_UNAPPROVED_PATH',
            unapprovedFiles: whitelistResult.unapprovedFiles
          }
        }
      })
    }
    return cpy
  }

  cpy.lastAuditError = ''
  if (bal.slv) {
    bal.slv({
      difBit: {
        idx: 'audit-pr-diff-ok',
        val: 1,
        dat: { valid: true }
      }
    })
  }

  return cpy
}
