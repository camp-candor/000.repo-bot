import type { FencingModel } from '../fencing.model.js'
import type FencingBit from '../fce/fencing.bit.js'
import type { TaskLeaseRecord, EpochValidationResult } from '../fce/fencing.bit.js'
import type State from '../../99.core/state.js'

const DEFAULT_LEASE_TTL_MS = 180000 // 3 minutes

export const initFencing = (cpy: FencingModel, bal: FencingBit, ste: State) => {
  if (bal.slv) bal.slv({ fncBit: { idx: 'init-fencing', dat: cpy } })
  return cpy
}

export const advanceTaskEpoch = (
  activeLeases: Record<string, TaskLeaseRecord>,
  taskId: string,
  workerId?: string,
  ttlMs: number = DEFAULT_LEASE_TTL_MS
): TaskLeaseRecord => {
  const currentLease = activeLeases[taskId]
  const nextEpoch = currentLease ? currentLease.epochToken + 1 : 1
  const leasedAt = Date.now()
  const expiresAt = leasedAt + ttlMs

  const newLease: TaskLeaseRecord = {
    taskId,
    epochToken: nextEpoch,
    leasedAt,
    expiresAt,
    workerId: workerId || 'default-worker'
  }

  activeLeases[taskId] = newLease
  return newLease
}

export const assertEpochLeaseValidity = (
  activeLeases: Record<string, TaskLeaseRecord>,
  taskId: string,
  incomingEpoch: number,
  currentTimeMs: number = Date.now()
): EpochValidationResult => {
  const currentLease = activeLeases[taskId]

  if (!currentLease) {
    return {
      valid: false,
      reason: 'NO_ACTIVE_LEASE_FOUND_FOR_TASK',
      presentedEpoch: incomingEpoch
    }
  }

  if (incomingEpoch < currentLease.epochToken) {
    return {
      valid: false,
      reason: 'ZOMBIE_EPOCH_MUTATION_REJECTED',
      activeEpoch: currentLease.epochToken,
      presentedEpoch: incomingEpoch
    }
  }

  if (incomingEpoch > currentLease.epochToken) {
    return {
      valid: false,
      reason: 'FUTURE_EPOCH_TOKEN_UNRECOGNIZED',
      activeEpoch: currentLease.epochToken,
      presentedEpoch: incomingEpoch
    }
  }

  if (currentTimeMs > currentLease.expiresAt) {
    return {
      valid: false,
      reason: 'TASK_LEASE_EXPIRED',
      activeEpoch: currentLease.epochToken,
      presentedEpoch: incomingEpoch
    }
  }

  return {
    valid: true,
    activeEpoch: currentLease.epochToken,
    presentedEpoch: incomingEpoch
  }
}

export const formatFencingDiagnostic = (
  valid: boolean,
  reason?: string,

  presentedEpoch?: number

  return `>> [FENCING] [FAIL] Reason: ${reason || 'UNKNOWN_FENCING_CONFLICT'} :: Presented: ${presentedEpoch} :: Active: ${activeEpoch}`
}

export const acquireTaskEpoch = async (
  cpy: FencingModel,
  bal: FencingBit,
  ste: State
) => {
  const taskId = bal.taskId || 'default-task'
  const ttlMs = bal.ttlMs || DEFAULT_LEASE_TTL_MS

  const newLease = advanceTaskEpoch(cpy.activeLeases, taskId, bal.workerId, ttlMs)
  cpy.totalLeasesIssued += 1
  cpy.lastLeaseTimestamp = newLease.leasedAt

  if (bal.slv) {
    bal.slv({
      fncBit: {
        idx: 'acquire-task-epoch-ok',
        val: newLease.epochToken,
        dat: {
          lease: newLease,
          diagnostic: formatFencingDiagnostic(true, undefined, newLease.epochToken)
        }
      }
    })
  }

  return cpy
}

export const validateEpochLease = async (
  cpy: FencingModel,
  bal: FencingBit,
  ste: State
) => {
  const taskId = bal.taskId || 'default-task'
  const incomingEpoch = bal.incomingEpoch ?? 0

  const validation = assertEpochLeaseValidity(cpy.activeLeases, taskId, incomingEpoch)

  if (!validation.valid) {
    cpy.totalZombieRejections += 1
    cpy.lastRejectReason = validation.reason!
    if (bal.slv) {
      bal.slv({
        fncBit: {
          idx: 'validate-epoch-lease-fail',
          val: 0,
          dat: {
            valid: false,
            reason: validation.reason,
            activeEpoch: validation.activeEpoch,
            presentedEpoch: validation.presentedEpoch,
            diagnostic: formatFencingDiagnostic(
              false,
              validation.reason,
              validation.activeEpoch,
              validation.presentedEpoch
            )
          }
        }
      })
    }
    return cpy
  }

  if (bal.slv) {
    bal.slv({
      fncBit: {
        idx: 'validate-epoch-lease-ok',
        val: 1,
        dat: {
          valid: true,
          activeEpoch: validation.activeEpoch,
          presentedEpoch: validation.presentedEpoch,
          diagnostic: formatFencingDiagnostic(true, undefined, validation.activeEpoch)
        }
      }
    })
  }

  return cpy
}
