const MAX_FILE_BLAST_RADIUS = 50

const PROTECTED_INFRASTRUCTURE_PATTERNS = [
  /^\.github\/workflows\/.+/,
  /^tests\/.+/,
  /^vitest\.config\.ts$/,
  /^apps\/995\.library\/995\.library\/.+/,

export const normalizePath = (pathStr: string): string => {
  return pathStr.replace(/\\/g, '/').replace(/^\/+/, '').trim()
}

export const evaluateBlastRadius = (
  files: DiffFileEntry[]
): { valid: boolean; reason?: string } => {
  if (files.length > MAX_FILE_BLAST_RADIUS) {
    return {
      valid: false,
      reason: 'BLAST_RADIUS_EXCEEDED_MAX_50_FILES'
    }
  }
  return { valid: true }
}

export const evaluateProtectedPaths = (
  files: DiffFileEntry[]
): { valid: boolean; offendingPath?: string; reason?: string } => {
  for (const entry of files) {
    const rawPath = entry.filename

    if (
      rawPath.includes('../') ||
      rawPath.includes('..\\') ||
      rawPath.includes('%2e%2e') ||
      rawPath.includes('\0')
    ) {
      return {
        valid: false,
        offendingPath: rawPath,
        reason: 'DIRECTORY_TRAVERSAL_DETECTED'
      }
    }

    const normalized = normalizePath(rawPath)

    for (const pattern of PROTECTED_INFRASTRUCTURE_PATTERNS) {
      if (pattern.test(normalized)) {
        return {
          valid: false,
          offendingPath: normalized,
          reason: 'PROTECTED_INFRASTRUCTURE_MUTATION_BREACH'
        }
      }
    }
  }

  return { valid: true }
}

export const evaluateWhitelistConformance = (
  files: DiffFileEntry[],
  whitelist?: string[]
): WhitelistConformanceResult => {
  if (!whitelist || whitelist.length === 0) {
    return { valid: true }
  }

  const unapproved: string[] = []

  for (const entry of files) {
    const normalized = normalizePath(entry.filename)
    const allowed = whitelist.some((rule) => {
      const cleanRule = normalizePath(rule)
      if (cleanRule.endsWith('/**')) {
        const prefix = cleanRule.slice(0, -3)
        return normalized.startsWith(prefix)
      }
      if (cleanRule.endsWith('/*')) {
        const prefix = cleanRule.slice(0, -2)
        return (
          normalized.startsWith(prefix) &&
          !normalized.slice(prefix.length + 1).includes('/')
        )
      }
      return normalized === cleanRule
    })

    if (!allowed) {
      unapproved.push(normalized)
    }
  }

  if (unapproved.length > 0) {
    return {
      valid: false,
      reason: 'SCOPE_VIOLATION_UNAPPROVED_PATH',
      unapprovedFiles: unapproved
    }
  }

  return { valid: true }
}

export const formatDiffAuditDiagnostic = (
  valid: boolean,
  reason?: string,
  unapprovedFiles?: string[]
): string => {
  if (valid) {
    return '>> [SCOPE_CHECK] [OK] Diff conforms strictly to task whitelist'

  let diagnostic = `>> [SCOPE_CHECK] [FAIL] Reason: ${reason || 'UNKNOWN_SCOPE_BREACH'}`
  if (unapprovedFiles && unapprovedFiles.length > 0) {
    diagnostic += ` :: Offending Files: [${unapprovedFiles.join(', ')}]`
  }
  return diagnostic
}

              whitelistResult.unapprovedFiles
            )
          }
        }
      })
    }
    return cpy
  }

  // Audit Passed
  cpy.lastAuditError = ''
  cpy.lastUnapprovedFiles = []

    bal.slv({
      difBit: {
        idx: 'audit-pr-diff-ok',
        val: 1,
        dat: {
          valid: true,
          diagnostic: formatDiffAuditDiagnostic(true)
        }
      }
    })
  }

  return cpy
}

  model: DiffModel = new DiffModel(),

) {
