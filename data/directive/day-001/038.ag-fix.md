```typescript
/**
 * Sanitizes a Slack channel target by stripping whitespace and wrapping quotes.
 * Defaults to the canonical alphanumeric ID of #ops-bridge if unset or provided as raw text.
 */
export function sanitizeChannelId(raw?: string): string {
  const cleaned = (raw || '')
    .trim()
    .replace(/^["']|["']$/g, '')
    .trim()
  if (!cleaned || cleaned === '#ops-bridge') {
    return 'C0C40FMRQ9H'
  }
  return cleaned
}
```
