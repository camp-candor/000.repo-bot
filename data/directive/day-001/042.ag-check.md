### IV. SYNTHETIC VERIFICATION SCRIPT (STORAGE SUB-MENU SIMULATION)

Execute this script via vitest or Node to simulate the sub-menu actions in an
isolated runtime:

```typescript
import { describe, it, expect, vi } from 'vitest'
import {
  checkDrainageStatus,
  countdownDrainage,
  fetchStorageRecords,
  inspectStorageRecord,
} from './packages/132.github/03.storage.unit/buz/storage.buzz.js'
import { StorageModel } from './packages/132.github/03.storage.unit/storage.model.js'

describe('Synthetic Storage Sub-Menu Integration Verification', () => {
  it('executes the full forensic sub-menu lifecycle cleanly', async () => {
    const consoleOutputs: string[] = []
    ;(global as any).LIBRARY = {
      hunt: vi.fn().mockImplementation(async (action: string, payload: any) => {
        if (action.includes('Update Console') && payload?.src) {
          consoleOutputs.push(payload.src)
        }
        return {}
      }),
    }

    const model = new StorageModel()

    // 1. Verify Status & Recency
    global.fetch = vi.fn().mockResolvedValueOnce(
      new Response(
        JSON.stringify({
          ok: true,
          serverTime: 1727203600000,
          lastDrainedAt: 1727200000000,
          undrainedCount: 15,
          totalHotRecords: 60,
          cronSchedule: '0 0 * * * (Midnight UTC)',
        }),
        { status: 200 },
      ),
    ) as any

    const slv1 = vi.fn()
    await checkDrainageStatus(model, { idx: 'check', slv: slv1 }, {} as any)
    expect(slv1).toHaveBeenCalledWith(
      expect.objectContaining({ strBit: expect.objectContaining({ val: 1 }) }),
    )
    expect(
      consoleOutputs.some((line) =>
        line.includes('Elapsed Since Sync: 60 minutes ago'),
      ),
    ).toBe(true)

    // 2. Verify Countdown Calculation
    const slv2 = vi.fn()
    await countdownDrainage(model, { idx: 'countdown', slv: slv2 }, {} as any)
    expect(slv2).toHaveBeenCalled()
    const countdownDat = slv2.mock.calls[0][0].strBit.dat
    expect(countdownDat.hours).toBeGreaterThanOrEqual(0)
    expect(countdownDat.hours).toBeLessThanOrEqual(24)

    // 3. Verify 100-Record Fetch & Forensic Record Inspection
    const mockRecord = {
      sequence_id: 101,
      created_at: 1727203600000,
      repository: 'astro-kahn-it-com/001.goblin-lore',
      task_id: 'TASK-31862',
      event_type: 'CAS_MERGE',
      actor_id: 'elliotbradly',
      head_sha: 'dab76e8',
      prev_hash: '0000genesisprevhash',
      record_hash: '1111recordhash',
      payload_json: JSON.stringify({
        title: 'chore: bump main package version',
        commit: { message: 'chore: bump version to 0.0.1' },
      }),
      drained_at: null,
    }

    global.fetch = vi.fn().mockResolvedValueOnce(
      new Response(JSON.stringify({ ok: true, events: [mockRecord] }), {
        status: 200,
      }),
    ) as any

    const slv3 = vi.fn()
    await fetchStorageRecords(
      model,
      { idx: 'fetch', val: 100, slv: slv3 },
      {} as any,
    )
    expect(slv3).toHaveBeenCalled()
    expect(slv3.mock.calls[0][0].strBit.lst.length).toBe(1)

    const slv4 = vi.fn()
    await inspectStorageRecord(
      model,
      { idx: 'inspect', dat: mockRecord, slv: slv4 },
      {} as any,
    )
    expect(slv4).toHaveBeenCalledWith(
      expect.objectContaining({ strBit: expect.objectContaining({ val: 1 }) }),
    )
    expect(
      consoleOutputs.some((line) =>
        line.includes('Commit Message: "chore: bump main package version"'),
      ),
    ).toBe(true)
    expect(
      consoleOutputs.some((line) =>
        line.includes('Storage State : [HOT BUFFER IN D1]'),
      ),
    ).toBe(true)
  })
})
```
