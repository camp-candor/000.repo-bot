7. **IDEMPOTENT COMPLETION RPC:** Network latency or dropped TCP connections
   must never cause spurious `HTTP 409 Conflict` errors on retried promotion
   requests. If the DO receives a promotion RPC for an already-promoted asset
   matching the same epoch and content-addressed key, it must return
   `HTTP 200 OK` immediately without re-executing state transitions.

### II. ARCHITECTURAL REQUIREMENTS

#### 1. Content-Addressed Staging Schema Engine (`apps/worker/src/storage/r2Promotion.ts`)

- **Staging Key Format:**
  `staging/renders/${taskId}/epoch_${epoch}_${sha256}.${extension}`

- Example:
  `staging/renders/shot-042/epoch_1_a8f5b3c2e1d0f4a8b7c9e0d1f2a3b4c5.mp4`

- **Canonical Release Key Format:**
  `canonical/renders/${taskId}/${sha256}.${extension}`

- **Key Validation & Parsing Utilities:**
- `parseStagingKey(key: string)`: Extracts `taskId`, `epoch`, `sha256`, and
  `extension` using strict regex matching. Throws if malformed.
- `validateStagingKey(key: string, expectedTaskId: string, expectedEpoch: number)`:
  Asserts that the key structurally belongs to the expected task and epoch.

- `buildCanonicalKey(taskId: string, sha256: string, extension: string)`:
  Constructs the canonical object reference.

- **Manifest Data Structure (`CanonicalManifestEntry`):**
- `taskId: string`

- `epoch: number`

- `stagingKey: string`

- `canonicalKey: string`

- `sha256: string`

- `workerId: string`

- `promotedAtMs: number`

#### 2. Atomic Promotion Gate & Zombie Fencing (`apps/worker/src/objects/ShotCoordinatorDO.ts`)

- **Ingress Completion Endpoint (`POST /tasks/complete`):**
- Payload:
  `{ workerId: string, epoch: number, stagingKey: string, sha256: string, metadata?: Record<string, any> }`.

- **Idempotency & Replay Evaluation:**
- Check if `this.canonicalManifest?.stagingKey === stagingKey` and
  `this.canonicalManifest?.epoch === epoch`.

- If matched: return
  `{ ok: true, fsmState: this.fsmState, epoch: this.currentEpoch, alreadyPromoted: true }`
  with `HTTP 200 OK`.

- **Monotonic Epoch Fencing Check:**
- Compare `epoch === this.currentEpoch`.

- If `epoch < this.currentEpoch`: Immediately reject with `HTTP 409 Conflict`
  and payload
  `{ error: 'STALE_EPOCH_ZOMBIE_REJECTED', presentedEpoch: epoch, activeEpoch: this.currentEpoch }`.

- **Staging Key Verification:**
- Execute `validateStagingKey(stagingKey, this.taskId, this.currentEpoch)`.

- If invalid: reject with `HTTP 400 Bad Request`.

- **Atomic Storage Promotion:**
- Construct `CanonicalManifestEntry`.

- Transition FSM state from `RUNNING` to `VERIFYING` (which unconditionally
  deletes the hardware watchdog alarm).

- Atomically persist `this.canonicalManifest` alongside `task_record` in
  `ctx.storage`.

- Log:
  `>> [PROMOTION:OK] Task '${taskId}' successfully promoted artifact '${stagingKey}' under Epoch ${epoch} [OK]`.

```typescript
export interface ParsedStagingKey {
  taskId: string
  epoch: number
  sha256: string
  extension: string
}

export interface CanonicalManifestEntry {
  taskId: string
  epoch: number
  stagingKey: string
  canonicalKey: string
  sha256: string
  workerId: string
  promotedAtMs: number
  metadata?: Record<string, any>
}

const STAGING_KEY_REGEX =
  /^staging\/renders\/([a-zA-Z0-9_\-]+)\/epoch_(\d+)_([a-fA-F0-9]{32,64})\.([a-zA-Z0-9]+)$/

/**
 * Parses an epoch-scoped, content-addressed staging key.
 */
export function parseStagingKey(key: string): ParsedStagingKey {
  const match = STAGING_KEY_REGEX.exec(key)
  if (!match) {
    throw new Error(
      `INVALID_STAGING_KEY: Key '${key}' does not match staging format 'staging/renders/{taskId}/epoch_{epoch}_{sha256}.{ext}'.`,
    )
  }

  return {
    taskId: match[1],
    epoch: parseInt(match[2], 10),
    sha256: match[3].toLowerCase(),
    extension: match[4].toLowerCase(),
  }
}

/**
 * Validates that a staging key matches the expected taskId and active lease epoch.
 */
export function validateStagingKey(
  key: string,
  expectedTaskId: string,
  expectedEpoch: number,
): ParsedStagingKey {
  const parsed = parseStagingKey(key)

  if (parsed.taskId !== expectedTaskId) {
    throw new Error(
      `TASK_ID_MISMATCH: Staging key belongs to '${parsed.taskId}', expected '${expectedTaskId}'.`,
    )
  }

  if (parsed.epoch !== expectedEpoch) {
    throw new Error(
      `EPOCH_TOKEN_MISMATCH: Staging key stamped with Epoch ${parsed.epoch}, active lease is Epoch ${expectedEpoch}.`,
    )
  }

  return parsed
}

/**
 * Generates the canonical storage path for an approved render.
 */
export function buildCanonicalKey(
  taskId: string,
  sha256: string,
  extension: string,
): string {
  return `canonical/renders/${taskId}/${sha256}.${extension}`
}
```

```typescript
import {
  evaluateFsmTransition,
  isRevokingEvent,
  isTerminalState,
  type TaskState,
  type TaskEvent,
  type TransitionContext,
} from '../fsm/transitions.js'
import {
  scheduleTimer,
  cancelTimer,
  getEarliestFireTime,
  pollDueTimers,
  type ScheduledTimer,
} from './timerScheduler.js'
import {
  validateStagingKey,
  buildCanonicalKey,
  type CanonicalManifestEntry,
} from '../storage/r2Promotion.js'

export const WATCHDOG_WINDOW_MS = 30_000
export const MAX_EXECUTION_CAP_MS = 15 * 60_000

export interface TaskRecord {
  taskId: string
  fsmState: TaskState
  currentEpoch: number
  leaseExpiresAt: number
  attemptCount: number
  maxAttempts: number
  activeWorkerId: string | null
  isHighRisk: boolean
  timers: ScheduledTimer[]
  canonicalManifest: CanonicalManifestEntry | null
  updatedAt: number
}

export class ShotCoordinatorDO {
  private taskId = ''
  private fsmState: TaskState = 'PENDING'
  private currentEpoch = 0
  private leaseExpiresAt = 0
  private attemptCount = 0
  private maxAttempts = 3
  private activeWorkerId: string | null = null
  private isHighRisk = false
  private timers: ScheduledTimer[] = []
  private canonicalManifest: CanonicalManifestEntry | null = null
  private updatedAt = Date.now()

  constructor(
    private ctx: any,
    private env: any,
  ) {
    this.ctx.blockConcurrencyWhile?.(async () => {
      const record = await this.ctx.storage.get('task_record')
      if (record && typeof record === 'object') {
        this.taskId = record.taskId || ''
        this.fsmState = record.fsmState || 'PENDING'
        this.currentEpoch = Number(record.currentEpoch || 0)
        this.leaseExpiresAt = Number(record.leaseExpiresAt || 0)
        this.attemptCount = Number(record.attemptCount || 0)
        this.maxAttempts = Number(record.maxAttempts || 3)
        this.activeWorkerId = record.activeWorkerId || null
        this.isHighRisk = Boolean(record.isHighRisk)
        this.timers = Array.isArray(record.timers) ? record.timers : []
        this.canonicalManifest = record.canonicalManifest || null
        this.updatedAt = Number(record.updatedAt || Date.now())
        console.log(
          `>> [DO:BOOT] Task '${this.taskId}' loaded. Epoch: ${this.currentEpoch}, Manifest: ${this.canonicalManifest ? 'PRESENT' : 'NONE'} [OK]`,
        )
      }
    })
  }

  async applyTransition(
    event: TaskEvent,
    actor: { type: string; id: string; epoch?: number },
    payload?: any,
  ): Promise<TaskRecord> {
    const context: TransitionContext = {
      attemptCount: this.attemptCount,
      maxAttempts: this.maxAttempts,
      isHighRisk: this.isHighRisk,
    }

    const result = evaluateFsmTransition(this.fsmState, event, context)
    if (!result.isValid) {
      console.error(`>> [FSM:REJECT] ${result.error} [FAIL]`)
      throw new Error(result.error)
    }

    if (result.isTerminalNoOp) {
      return this.getSnapshot()
    }

    const previousState = this.fsmState
    const nextState = result.nextState

    if (previousState === nextState) {
      return this.getSnapshot()
    }

    const now = Date.now()

    if (
      event === 'CLAIM_LEASE' ||
      isRevokingEvent(event) ||
      event === 'RETRY'
    ) {
      this.currentEpoch += 1
      console.log(
        `>> [EPOCH:BUMP] Advanced epoch to ${this.currentEpoch} via '${event}' [OK]`,
      )
    }

    this.fsmState = nextState
    this.updatedAt = now

    if (event === 'CLAIM_LEASE') {
      this.attemptCount += 1
      this.activeWorkerId = actor.id || null
      this.leaseExpiresAt = now + WATCHDOG_WINDOW_MS
    }

    if (nextState === 'RUNNING') {
      this.timers = scheduleTimer(this.timers, {
        id: 'watchdog',
        type: 'WATCHDOG',
        fireAt: now + WATCHDOG_WINDOW_MS,
        epoch: this.currentEpoch,
        epochScoped: true,
      })
      this.timers = scheduleTimer(this.timers, {
        id: 'absolute_cap',
        type: 'ABSOLUTE_CAP',
        fireAt: now + MAX_EXECUTION_CAP_MS,
        epoch: this.currentEpoch,
        epochScoped: true,
      })
      await this.syncHardwareAlarm()
    } else if (previousState === 'RUNNING' && nextState !== 'RUNNING') {
      this.timers = cancelTimer(this.timers, 'watchdog')
      this.timers = cancelTimer(this.timers, 'absolute_cap')
      await this.syncHardwareAlarm()
    }

    if (isTerminalState(nextState)) {
      this.timers = []
      await this.ctx.storage.deleteAlarm()
      this.activeWorkerId = null
      this.leaseExpiresAt = 0
    }

    const snapshot = this.getSnapshot()
    await this.ctx.storage.put('task_record', snapshot)
    console.log(
      `>> [FSM:TRANSITION] '${previousState}' -> '${nextState}' via '${event}' (Epoch: ${this.currentEpoch}) [OK]`,
    )
    return snapshot
  }

  async alarm(): Promise<void> {
    const now = Date.now()
    console.warn(
      `>> [ALARM:WAKE] Watchdog alarm triggered for task '${this.taskId}' at ${now} [ALERT]`,
    )

    const { due, remaining } = pollDueTimers(
      this.timers,
      now,
      this.currentEpoch,
    )
    this.timers = remaining

    for (const timer of due) {
      if (
        timer.type === 'WATCHDOG' &&
        (this.fsmState === 'RUNNING' || this.fsmState === 'CLAIMED')
      ) {
        console.warn(
          `>> [WATCHDOG:EXPIRE] Missed pulses. Revoking lease for task '${this.taskId}' [ALERT]`,
        )
        await this.applyTransition('WATCHDOG_EXPIRE', {
          type: 'SYSTEM',
          id: 'WATCHDOG_ALARM',
        })
      } else if (timer.type === 'ABSOLUTE_CAP' && this.fsmState === 'RUNNING') {
        console.error(
          `>> [ABSOLUTE_CAP:EXPIRE] Max 15m execution exceeded for task '${this.taskId}' [FAIL]`,
        )
        await this.applyTransition('WATCHDOG_EXPIRE', {
          type: 'SYSTEM',
          id: 'ABSOLUTE_CAP_ALARM',
        })
      }
    }

    await this.syncHardwareAlarm()
    await this.ctx.storage.put('task_record', this.getSnapshot())
  }

  private async syncHardwareAlarm(): Promise<void> {
    const earliest = getEarliestFireTime(this.timers)
    if (earliest !== null) {
      await this.ctx.storage.setAlarm(earliest)
    } else {
      await this.ctx.storage.deleteAlarm()
    }
  }

  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url)

    try {
      if (request.method === 'POST' && url.pathname.endsWith('/leases/claim')) {
        const body = (await request.json().catch(() => ({}))) as any
        const workerId = body.workerId || 'UNKNOWN_WORKER'

        if (this.fsmState !== 'PENDING') {
          return new Response(
            JSON.stringify({
              error: `TASK_UNAVAILABLE: State is '${this.fsmState}'`,
            }),
            { status: 409, headers: { 'Content-Type': 'application/json' } },
          )
        }

        if (!this.taskId) {
          this.taskId = body.taskId || 'task-default'
          this.isHighRisk = Boolean(body.isHighRisk)
        }

        const record = await this.applyTransition('CLAIM_LEASE', {
          type: 'WORKER',
          id: workerId,
        })

        return new Response(
          JSON.stringify({
            ok: true,
            taskId: record.taskId,
            epoch: record.currentEpoch,
            leaseExpiresAt: record.leaseExpiresAt,
            fsmState: record.fsmState,
          }),
          { status: 200, headers: { 'Content-Type': 'application/json' } },
        )
      }

      if (
        request.method === 'POST' &&
        url.pathname.endsWith('/leases/acknowledge')
      ) {
        const body = (await request.json().catch(() => ({}))) as any
        const { workerId, epoch } = body

        if (epoch !== this.currentEpoch) {
          return new Response(
            JSON.stringify({
              error: `STALE_EPOCH: Presented ${epoch}, Active ${this.currentEpoch}`,
            }),
            { status: 409, headers: { 'Content-Type': 'application/json' } },
          )
        }

        const record = await this.applyTransition('ACK_RUNNING', {
          type: 'WORKER',
          id: workerId,
        })

        return new Response(
          JSON.stringify({
            ok: true,
            fsmState: record.fsmState,
            epoch: record.currentEpoch,
          }),
          { status: 200, headers: { 'Content-Type': 'application/json' } },
        )
      }

      if (
        request.method === 'POST' &&
        url.pathname.endsWith('/leases/heartbeat')
      ) {
        const body = (await request.json().catch(() => ({}))) as any
        const { epoch } = body
        const now = Date.now()

        if (epoch !== this.currentEpoch) {
          console.warn(
            `>> [HEARTBEAT:REJECT] Stale Epoch ${epoch} != Active ${this.currentEpoch} [FAIL]`,
          )
          return new Response(
            JSON.stringify({
              error: `HTTP 409 Conflict: Stale Epoch ${epoch}. Active is ${this.currentEpoch}`,
            }),
            { status: 409, headers: { 'Content-Type': 'application/json' } },
          )
        }

        if (this.fsmState !== 'RUNNING') {
          return new Response(
            JSON.stringify({
              error: `INVALID_STATE: Task is in '${this.fsmState}', expected RUNNING`,
            }),
            { status: 409, headers: { 'Content-Type': 'application/json' } },
          )
        }

        this.leaseExpiresAt = now + WATCHDOG_WINDOW_MS
        this.updatedAt = now

        this.timers = scheduleTimer(this.timers, {
          id: 'watchdog',
          type: 'WATCHDOG',
          fireAt: this.leaseExpiresAt,
          epoch: this.currentEpoch,
          epochScoped: true,
        })

        await this.syncHardwareAlarm()
        await this.ctx.storage.put('task_record', this.getSnapshot())

        return new Response(
          JSON.stringify({
            ok: true,
            epoch: this.currentEpoch,
            leaseExpiresAt: this.leaseExpiresAt,
          }),
          { status: 200, headers: { 'Content-Type': 'application/json' } },
        )
      }

      // PROMOTION BOUNDARY & ZOMBIE FENCING GATE
      if (
        request.method === 'POST' &&
        url.pathname.endsWith('/tasks/complete')
      ) {
        const body = (await request.json().catch(() => ({}))) as any
        const { workerId, epoch, stagingKey, sha256, metadata } = body

        // 1. Idempotency Check: Safe replay for already-promoted tasks
        if (
          this.canonicalManifest &&
          this.canonicalManifest.stagingKey === stagingKey &&
          this.canonicalManifest.epoch === epoch
        ) {
          console.log(
            `>> [PROMOTION:IDEMPOTENT] Replay request for already-promoted artifact '${stagingKey}' [OK]`,
          )
          return new Response(
            JSON.stringify({
              ok: true,
              fsmState: this.fsmState,
              epoch: this.currentEpoch,
              alreadyPromoted: true,
              canonicalKey: this.canonicalManifest.canonicalKey,
            }),
            { status: 200, headers: { 'Content-Type': 'application/json' } },
          )
        }

        // 2. Monotonic Epoch Verification (Zombie Runner Defense)
        if (epoch !== this.currentEpoch) {
          console.warn(
            `>> [ZOMBIE:FENCED] Rejected promotion from zombie worker '${workerId}'. Presented Epoch: ${epoch}, Active Epoch: ${this.currentEpoch} [FAIL]`,
          )
          return new Response(
            JSON.stringify({
              error: `HTTP 409 Conflict: Stale Epoch ${epoch}. Active Epoch is ${this.currentEpoch}`,
              fenced: true,
            }),
            { status: 409, headers: { 'Content-Type': 'application/json' } },
          )
        }

        if (this.fsmState !== 'RUNNING') {
          return new Response(
            JSON.stringify({
              error: `INVALID_STATE: Task is in state '${this.fsmState}', expected RUNNING`,
            }),
            { status: 409, headers: { 'Content-Type': 'application/json' } },
          )
        }

        // 3. Staging Key Schema & Bound Validation
        let parsed
        try {
          parsed = validateStagingKey(
            stagingKey,
            this.taskId,
            this.currentEpoch,
          )
        } catch (valErr: any) {
          return new Response(
            JSON.stringify({
              error: valErr.message || 'MALFORMED_STAGING_KEY',
            }),
            { status: 400, headers: { 'Content-Type': 'application/json' } },
          )
        }

        // 4. Construct Canonical Manifest Entry
        const canonicalKey = buildCanonicalKey(
          this.taskId,
          parsed.sha256,
          parsed.extension,
        )
        this.canonicalManifest = {
          taskId: this.taskId,
          epoch: this.currentEpoch,
          stagingKey,
          canonicalKey,
          sha256: parsed.sha256,
          workerId: workerId || this.activeWorkerId || 'UNKNOWN',
          promotedAtMs: Date.now(),
          metadata: metadata || {},
        }

        // 5. Atomic Promotion: Advance FSM state to VERIFYING (disarming watchdog alarms)
        const record = await this.applyTransition(
          'SUBMIT_VERIFICATION',
          { type: 'WORKER', id: workerId },
          this.canonicalManifest,
        )

        console.log(
          `>> [PROMOTION:OK] Successfully promoted '${stagingKey}' -> '${canonicalKey}' under Epoch ${this.currentEpoch} [OK]`,
        )

        return new Response(
          JSON.stringify({
            ok: true,
            fsmState: record.fsmState,
            epoch: record.currentEpoch,
            promoted: true,
            canonicalKey,
          }),
          { status: 200, headers: { 'Content-Type': 'application/json' } },
        )
      }

      if (request.method === 'GET' && url.pathname.endsWith('/state')) {
        return new Response(JSON.stringify(this.getSnapshot()), {
          status: 200,
          headers: { 'Content-Type': 'application/json' },
        })
      }

      return new Response('NOT_FOUND', { status: 404 })
    } catch (err: any) {
      console.error(`>> [DO:ERROR] ${err.message || String(err)}`)
      return new Response(
        JSON.stringify({ error: err.message || 'INTERNAL_ERROR' }),
        {
          status: 500,
          headers: { 'Content-Type': 'application/json' },
        },
      )
    }
  }

  private getSnapshot(): TaskRecord {
    return {
      taskId: this.taskId,
      fsmState: this.fsmState,
      currentEpoch: this.currentEpoch,
      leaseExpiresAt: this.leaseExpiresAt,
      attemptCount: this.attemptCount,
      maxAttempts: this.maxAttempts,
      activeWorkerId: this.activeWorkerId,
      isHighRisk: this.isHighRisk,
      timers: [...this.timers],
      canonicalManifest: this.canonicalManifest
        ? { ...this.canonicalManifest }
        : null,
      updatedAt: this.updatedAt,
    }
  }
}
```
