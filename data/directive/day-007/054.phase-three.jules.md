### II. ARCHITECTURAL REQUIREMENTS

#### 1. Three-Tier Progressive Escalation Engine (`apps/worker/src/escalation/progressiveEscalation.ts`)

- **Escalation Tier Enumeration:**
- `TIER_1_MICRO_PROMPT`: Attempt 1 failure with convergence trajectory.
- `TIER_2_AST_RECONDITION`: Attempt 2 failure with convergence trajectory.
- `TIER_3_ARCHITECT_ESCALATION`: Attempt 3 failure OR any attempt exhibiting
  `DIVERGENCE`.

- **Evaluation Models:**
- `EscalationContext`:
  `{ taskId: string; candidateSha: string; attemptCount: number; maxAttempts: number; trajectory: TrajectoryMode; auditHistory: AuditCycleRecord[]; currentWorkflow?: ComfyWorkflow; currentPrompts: PromptPayload }`
- `EscalationDirective`:
- `tier`: `EscalationTier`
- `action`: `'RETRY_MICRO_PATCH' | 'RETRY_MACRO_AST' | 'FREEZE_AND_QUARANTINE'`
- `remediatedJobPayload?: Partial<ExecutionJob>`
- `dossier?: StoryArchitectDossier`
- `rationale`: String description of escalation policy decision.

- **Policy Evaluator (`evaluateEscalationPolicy`):**
- If `trajectory === 'DIVERGENCE'`: Force immediate promotion to
  `TIER_3_ARCHITECT_ESCALATION`.
- If `attemptCount === 1`: Route to `TIER_1_MICRO_PROMPT`.
- If `attemptCount === 2`: Route to `TIER_2_AST_RECONDITION`.
- If `attemptCount >= 3`: Route to `TIER_3_ARCHITECT_ESCALATION`.

#### 2. Story Architect Dossier Generator (`apps/worker/src/escalation/storyArchitectDossier.ts`)

- **Dossier Schema (`StoryArchitectDossier`):**
- `dossierId`: Canonical identifier (`dossier-${taskId}-${Date.now()}`).
- `taskId`: Target shot identifier.
- `candidateSha`: SHA-256 hash of failed plate.
- `status`: `'BLOCKED'`.
- `exhaustedAttempts`: Total cycles attempted.
- `escalationTrigger`: `'MAX_ATTEMPTS_EXHAUSTED' | 'STRUCTURAL_DIVERGENCE'`.
- `trajectory`: `'CONVERGENCE' | 'DIVERGENCE'`.
- `timeline`: Array of attempt audit summaries.
- `contradictionLog`: Detailed summary of structural failures and invariant
  violations.
- `ledgerHashSnapshot`: Current head block hash of `ledgerChain`.
- `clickUpPayload`: Structured ASCII payload for issue tracker hand-off.
- `generatedAtIso`: ISO timestamp.

- **Formatter (`formatDossierForChatOps`):**
- Generates a pure 7-bit ASCII markdown block containing full diagnostic context
  for human Story Architect review.

#### 3. Durable Object Wiring (`apps/worker/src/objects/ShotCoordinatorDO.ts`)

- **Triage and Remediation Router Integration:**
- On `/tasks/triage-and-remediate`:
- Evaluates candidate against `DriftGauntlet`.
- Passes outcome to `evaluateEscalationPolicy`.
- If `TIER_1_MICRO_PROMPT` or `TIER_2_AST_RECONDITION`:

- Disarms lease watchdog alarm.
- Marks state as `BLOCKED`/`DLQ`.
- Appends `TIER_3_QUARANTINED` to `ledgerChain`.
- Dispatches async alert with `StoryArchitectDossier` to `CHATOPS_WEBHOOK_URL`.

import type {
  AuditCycleRecord,
  TrajectoryMode,
} from '../triage/circuitBreaker.js'

export interface StoryArchitectDossier {
  dossierId: string
  taskId: string
  candidateSha: string
  status: 'BLOCKED'
  exhaustedAttempts: number
  escalationTrigger: 'MAX_ATTEMPTS_EXHAUSTED' | 'STRUCTURAL_DIVERGENCE'
  trajectory: TrajectoryMode
  timeline: string[]
  contradictionLog: string
  ledgerHashSnapshot: string
  clickUpPayload: {
    taskName: string
    priority: 'URGENT' | 'HIGH'
    tags: string[]
    markdownDescription: string
  }
  generatedAtIso: string
}

/**
 * Builds an immutable escalation dossier for human Story Architect triage.
 */
export function buildStoryArchitectDossier(
  taskId: string,
  candidateSha: string,
  history: AuditCycleRecord[],
  trajectory: TrajectoryMode,
  ledgerHeadHash: string,
): StoryArchitectDossier {
  const exhaustedAttempts = history.length
  const isDivergent = trajectory === 'DIVERGENCE'
  const escalationTrigger = isDivergent
    ? 'STRUCTURAL_DIVERGENCE'
    : 'MAX_ATTEMPTS_EXHAUSTED'

  const timeline = history.map((record) => {
    const tokens =
      record.errorTokens.length > 0 ? record.errorTokens.join(', ') : 'none'
    return `[Cycle ${record.attempt}]${record.classification} :: Tokens: [${tokens}] (${new Date(record.timestampMs).toISOString()})`
  })

  let contradictionLog = ''
  if (isDivergent) {
    contradictionLog = `Structural divergence detected: Generation oscillated across conflicting taxonomy categories: ${history.map((h) => h.classification).join(' -> ')}. Automated retry halted to preserve compute.`
  } else {
    contradictionLog = `Attempt budget exhausted across ${exhaustedAttempts} cycles without convergence. Last classification: ${history[history.length - 1]?.classification || 'UNKNOWN'}.`
  }

  const taskName = `[P0 DRIFT ESCALATION] ${taskId} - Structural Remediation Required`
  const priority = isDivergent ? 'URGENT' : 'HIGH'
  const tags = [
    'drift-gauntlet',
    'story-architect-triage',
    isDivergent ? 'divergence-conflict' : 'budget-exhausted',
  ]

  const markdownDescription = [
    '# Story Architect Escalation Dossier',
    `**Task ID:** ${taskId}`,
    `**Candidate Plate SHA:** \`${candidateSha}\``,
    `**Escalation Trigger:** ${escalationTrigger}`,
    `**Trajectory:** ${trajectory}`,
    `**Exhausted Attempts:** ${exhaustedAttempts}`,
    `**Ledger State Digest:** \`${ledgerHeadHash}\``,
    '',
    '## Contradiction Analysis',
    `> ${contradictionLog}`,
    '',
    '## Cycle Audit Timeline',
    timeline.map((line) => `- ${line}`).join('\n'),
    '',
    '## Operator Action Required',
    '1. Inspect candidate plate render in staging scratch storage.',
    '2. Review prompt seed basin and evaluate character model sheet constraints.',
    '3. Provide manual prompt override or approve modified canon anchor via ChatOps `/repo-bot override`.',
  ].join('\n')

  return {
    dossierId: `dossier-${taskId}-${Date.now()}`,
    taskId,
    candidateSha,
    status: 'BLOCKED',
    exhaustedAttempts,
    escalationTrigger,
    trajectory,
    timeline,
    contradictionLog,
    ledgerHashSnapshot: ledgerHeadHash,
    clickUpPayload: {
      taskName,
      priority,
      tags,
      markdownDescription,
    },
    generatedAtIso: new Date().toISOString(),
  }
}

/**
 * Formats the dossier into standardized 7-bit ASCII text for ChatOps webhooks.
 */
export function formatDossierForChatOps(
  dossier: StoryArchitectDossier,
): string {
  return [
    '```text',
    '================================================================================',
    '>> [ESCALATION:TIER_3] STORY ARCHITECT INTERVENTION REQUIRED',
    '================================================================================',
    `TASK ID:        ${dossier.taskId}`,
    `DOSSIER ID:     ${dossier.dossierId}`,
    `TRIGGER:        ${dossier.escalationTrigger}`,
    `TRAJECTORY:     ${dossier.trajectory}`,
    `CYCLES TRIED:   ${dossier.exhaustedAttempts}`,
    `CANDIDATE SHA:  ${dossier.candidateSha}`,
    `LEDGER HEAD:    ${dossier.ledgerHashSnapshot.substring(0, 16)}...`,
    '--------------------------------------------------------------------------------',
    'TIMELINE:',
    dossier.timeline.map((t) => `  * ${t}`).join('\n'),
    '--------------------------------------------------------------------------------',
    `CONTRADICTION:  ${dossier.contradictionLog}`,
    'STATUS:         EXECUTION FROZEN :: TASK QUARANTINED TO DLQ [HALT]',
    '================================================================================',
    '```',
  ].join('\n')
}

import type {
  AuditCycleRecord,
  TrajectoryMode,
} from '../triage/circuitBreaker.js'
import type { VlmAuditReport } from '../triage/visualScanTrace.js'
import type { ExecutionJob } from '../queue/fairQueue.js'
import type { ComfyWorkflow } from '../remediation/patchPipeline.js'
import {
  synthesizeRemediatedPrompt,
  type PromptPayload,
  type PromptPatchResult,
} from '../remediation/promptSynthesizer.js'
import {
  buildStoryArchitectDossier,
  type StoryArchitectDossier,
} from './storyArchitectDossier.js'

export type EscalationTier =
  | 'TIER_1_MICRO_PROMPT'
  | 'TIER_2_AST_RECONDITION'
  | 'TIER_3_ARCHITECT_ESCALATION'

export interface EscalationContext {
  taskId: string
  candidateSha: string
  attemptCount: number
  maxAttempts: number
  trajectory: TrajectoryMode
  auditHistory: AuditCycleRecord[]
  latestReport: VlmAuditReport
  currentWorkflow?: ComfyWorkflow
  currentJob: ExecutionJob
  ledgerHeadHash?: string
}

export interface EscalationDirective {
  tier: EscalationTier
  action: 'RETRY_MICRO_PATCH' | 'RETRY_MACRO_AST' | 'FREEZE_AND_QUARANTINE'
  remediatedJob?: ExecutionJob
  dossier?: StoryArchitectDossier
  rationale: string
}

export class ProgressiveEscalationEngine {
  /**
   * Evaluates progressive escalation policies and returns actionable directives.
   */
  public evaluateEscalation(context: EscalationContext): EscalationDirective {
    const {
      taskId,
      candidateSha,
      attemptCount,
      maxAttempts,
      trajectory,
      auditHistory,
      latestReport,
      currentJob,
      currentWorkflow,
      ledgerHeadHash = '0'.repeat(64),
    } = context

    // RULE 1: Immediate Divergence Short-Circuit Law
    // If consecutive attempts jump across structural categories, skip Tier 2 immediately!
    if (trajectory === 'DIVERGENCE') {
      console.error(
        `>> [ESCALATE:DIVERGE] Structural divergence detected for '${taskId}'. Bypassing Tier 2 -> Direct Tier 3 [HALT]`,
      )
      const dossier = buildStoryArchitectDossier(
        taskId,
        candidateSha,
        auditHistory,
        trajectory,
        ledgerHeadHash,
      )

      return {
        tier: 'TIER_3_ARCHITECT_ESCALATION',
        action: 'FREEZE_AND_QUARANTINE',
        dossier,
        rationale:
          'Structural divergence across categories: fast-failed to Tier 3 human quarantine.',
      }
    }

    // RULE 2: Attempt Budget Ceiling
    if (attemptCount >= maxAttempts) {
      console.error(
        `>> [ESCALATE:EXHAUST] Max attempts reached (${attemptCount}/${maxAttempts}) for '${taskId}'. Escalating to Tier 3 [HALT]`,
      )
      const dossier = buildStoryArchitectDossier(
        taskId,
        candidateSha,
        auditHistory,
        trajectory,
        ledgerHeadHash,
      )

      return {
        tier: 'TIER_3_ARCHITECT_ESCALATION',
        action: 'FREEZE_AND_QUARANTINE',
        dossier,
        rationale: `Exhausted ${maxAttempts} automated cycles without convergence.`,
      }
    }

    const rawPrompts: PromptPayload = {
      positivePrompt: currentJob.prompts?.positive || '',
      negativePrompt: currentJob.prompts?.negative || '',
      seed: currentJob.seeds?.[0] ?? 42,
      denoiseStrength: currentJob.prompts?.denoiseStrength ?? 1.0,
      extraParameters: currentJob.prompts?.extraParameters,
    }

    // RULE 3: Tier 1 Micro-Prompt Synthesis (Attempt 1)
    if (attemptCount === 1) {
      console.log(
        `>> [ESCALATE:TIER_1] Applying Tier 1 Micro-Prompt Synthesis for '${taskId}' [RETRY]`,
      )
      const patchResult = synthesizeRemediatedPrompt(
        rawPrompts,
        latestReport,
        attemptCount,
      )
      const nextAttempt = attemptCount + 1

      const remediatedJob: ExecutionJob = {
        ...currentJob,
        attemptCount: nextAttempt,
        seeds: [patchResult.patchedPrompts.seed ?? currentJob.seeds[0] + 1013],
        prompts: {
          ...currentJob.prompts,
          positive: patchResult.patchedPrompts.positivePrompt,
          negative: patchResult.patchedPrompts.negativePrompt,
          denoiseStrength: patchResult.patchedPrompts.denoiseStrength,
          lastRemediationDiff: `[Tier 1 Micro] ${patchResult.diffSummary}`,
          escalationTier: 'TIER_1_MICRO_PROMPT',
        },
      }

      return {
        tier: 'TIER_1_MICRO_PROMPT',
        action: 'RETRY_MICRO_PATCH',
        remediatedJob,
        rationale: `Applied Tier 1 seed perturbation and anchor injection (${patchResult.strategyApplied}).`,
      }
    }

    // RULE 4: Tier 2 Macro-Workflow AST Reconditioning (Attempt 2)
    if (attemptCount === 2) {
      console.log(
        `>> [ESCALATE:TIER_2] Applying Tier 2 Macro-Workflow AST Reconditioning for '${taskId}' [RETRY]`,
      )
      const patchResult = synthesizeRemediatedPrompt(
        rawPrompts,
        latestReport,
        attemptCount,
      )

      // In Tier 2, apply structural conditioning: clamp denoise further, adjust CFG, expand negatives
      const boostedDenoise = Math.max(
        0.35,
        (patchResult.patchedPrompts.denoiseStrength ?? 1.0) - 0.05,
      )
      let patchedWorkflow = currentWorkflow
        ? structuredClone(currentWorkflow)
        : undefined

      if (patchedWorkflow) {
        for (const node of Object.values(patchedWorkflow)) {
          if (
            node.class_type === 'KSampler' ||
            node.class_type === 'KSamplerAdvanced'
          ) {
            node.inputs.denoise = boostedDenoise
            node.inputs.cfg = Math.min(
              9.0,
              (Number(node.inputs.cfg) || 7.0) + 0.5,
            ) // Boost prompt guidance
            node.inputs.steps = Math.min(
              35,
              (Number(node.inputs.steps) || 25) + 5,
            ) // Add refinement steps
          }
        }
      }

      const nextAttempt = attemptCount + 1
      const remediatedJob: ExecutionJob = {
        ...currentJob,
        attemptCount: nextAttempt,
        seeds: [patchResult.patchedPrompts.seed ?? currentJob.seeds[0] + 2048],
        prompts: {
          ...currentJob.prompts,
          positive: patchResult.patchedPrompts.positivePrompt,
          negative: patchResult.patchedPrompts.negativePrompt,
          denoiseStrength: boostedDenoise,
          workflowAst: patchedWorkflow || currentJob.prompts?.workflowAst,
          lastRemediationDiff: `[Tier 2 Macro AST] Denoise clamped to ${boostedDenoise}, CFG boosted, Sampler steps refined.`,
          escalationTier: 'TIER_2_AST_RECONDITION',
        },
      }

      return {
        tier: 'TIER_2_AST_RECONDITION',
        action: 'RETRY_MACRO_AST',
        remediatedJob,
        rationale:
          'Applied Tier 2 macro-workflow AST reconditioning (CFG boost, denoise clamp, sampler step refinement).',
      }
    }

    // Fallback default (Guaranteed termination)
    const fallbackDossier = buildStoryArchitectDossier(
      taskId,
      candidateSha,
      auditHistory,
      trajectory,
      ledgerHeadHash,
    )

    return {
      tier: 'TIER_3_ARCHITECT_ESCALATION',
      action: 'FREEZE_AND_QUARANTINE',
      dossier: fallbackDossier,
      rationale: 'Fallback quarantine triggered.',
    }
  }
}

import { FairQueue, type ExecutionJob } from '../queue/fairQueue.js'
import {
  createGenesisBlock,
  appendLedgerEntry,
  type LedgerEntry,
} from '../ledger/cryptoLedger.js'
import { formatDlqAlert, ChatOpsGateway } from '../telemetry/chatOpsGateway.js'
import { DriftGauntlet, type CandidatePlate } from '../triage/driftGauntlet.js'
import { type ComfyWorkflow } from '../remediation/patchPipeline.js'
import type { AuditCycleRecord } from '../triage/circuitBreaker.js'
import {
  ProgressiveEscalationEngine,
  type EscalationDirective,
} from '../escalation/progressiveEscalation.js'
import { formatDossierForChatOps } from '../escalation/storyArchitectDossier.js'

export class ShotCoordinatorDO {
  private fairQueue = new FairQueue()
  private activeJob: ExecutionJob | null = null
  private currentEpoch = 0
  private lastArtistId: string | null = null
  private ledgerChain: LedgerEntry[] = []
  private auditHistory: Map<string, AuditCycleRecord[]> = new Map()
  private driftGauntlet = new DriftGauntlet()
  private escalationEngine = new ProgressiveEscalationEngine()
  private chatOpsGateway = new ChatOpsGateway()

  constructor(
    private ctx: any,
    private env: any,
  ) {
    this.ctx.blockConcurrencyWhile?.(async () => {
      const queueData = await this.ctx.storage.get('pending_queue')
      if (Array.isArray(queueData)) this.fairQueue = new FairQueue(queueData)

      const active = await this.ctx.storage.get('active_job')
      if (active) this.activeJob = active

      const epoch = await this.ctx.storage.get('current_epoch')
      if (typeof epoch === 'number') this.currentEpoch = epoch

      const chain = await this.ctx.storage.get('ledger_chain')
      if (Array.isArray(chain)) this.ledgerChain = chain

      const history = await this.ctx.storage.get('audit_history')
      if (history && typeof history === 'object') {
        this.auditHistory = new Map(Object.entries(history))
      }

      console.log(
        `>> [DO:BOOT] ShotCoordinatorDO loaded. Epoch: ${this.currentEpoch}, Queue Depth: ${this.fairQueue.size()} [OK]`,
      )
    })
  }

  async alarm(): Promise<void> {
    console.warn(
      `>> [WATCHDOG:TRIP] Lease timeout for task '${this.activeJob?.taskId || 'NONE'}' [ALERT]`,
    )
    if (!this.activeJob) {
      await this.ctx.storage.deleteAlarm()
      return
    }

    const task = this.activeJob
    task.attemptCount += 1
    this.currentEpoch += 1

    if (task.attemptCount < task.maxAttempts) {
      this.fairQueue.requeue(task)
      this.activeJob = null
      this.ledgerChain = await appendLedgerEntry(
        this.ledgerChain,
        'WATCHDOG_TIMEOUT_REQUEUE',
        {
          taskId: task.taskId,
          attempt: task.attemptCount,
          newEpoch: this.currentEpoch,
        },
      )
    } else {
      this.activeJob = null
      this.ledgerChain = await appendLedgerEntry(
        this.ledgerChain,
        'CIRCUIT_BREAKER_DLQ',
        {
          taskId: task.taskId,
          attemptCount: task.attemptCount,
        },
      )
      if (this.env?.CHATOPS_WEBHOOK_URL) {
        const dlqAlert = formatDlqAlert(
          task.taskId,
          task.attemptCount,
          'MAX_ATTEMPTS_EXHAUSTED',
        )
        this.chatOpsGateway
          .dispatchAlert(this.env.CHATOPS_WEBHOOK_URL, dlqAlert)
          .catch(() => {})
      }
    }

    await this.ctx.storage.deleteAlarm()
    await this.persistState()
  }

  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url)

    try {
      // Task Enqueue Route
      if (request.method === 'POST' && url.pathname.endsWith('/enqueue')) {
        const body = (await request.json().catch(() => ({}))) as any
        const {
          taskId,
          artistId,
          idempotencyKey,
          workflowTemplate = 'default.json',
          prompts = {},
          seeds = [42],
          maxAttempts = 3,
        } = body

        const job: ExecutionJob = {
          taskId,
          artistId,
          idempotencyKey,
          workflowTemplate,
          prompts,
          seeds,
          enqueuedAt: Date.now(),
          attemptCount: 0,
          maxAttempts,
        }

        const enqueued = this.fairQueue.enqueue(job)
        if (!enqueued) {
          return new Response(JSON.stringify({ error: 'DUPLICATE_TASK' }), {
            status: 409,
          })
        }

        if (this.ledgerChain.length === 0) {
          this.ledgerChain = [
            await createGenesisBlock(taskId, { enqueuedAt: Date.now() }),
          ]
        } else {
          this.ledgerChain = await appendLedgerEntry(
            this.ledgerChain,
            'TASK_ENQUEUED',
            { taskId, artistId },
          )
        }

        await this.persistState()
        return new Response(
          JSON.stringify({ ok: true, taskId, position: this.fairQueue.size() }),
          { status: 200 },
        )
      }

      // Lease Claim Route
      if (request.method === 'POST' && url.pathname.endsWith('/leases/claim')) {
        if (this.activeJob !== null) {
          return new Response(JSON.stringify({ ok: false, busy: true }), {
            status: 200,
          })
        }

        const job = this.fairQueue.dequeueNextFair(this.lastArtistId)
        if (!job) return new Response(null, { status: 204 })

        this.currentEpoch += 1
        this.activeJob = job
        this.lastArtistId = job.artistId

        this.ledgerChain = await appendLedgerEntry(
          this.ledgerChain,
          'LEASE_CLAIMED',
          {
            taskId: job.taskId,
            epoch: this.currentEpoch,
          },
        )

        await this.ctx.storage.setAlarm(Date.now() + 30_000)
        await this.persistState()

        return new Response(
          JSON.stringify({
            ok: true,
            taskId: job.taskId,
            epoch: this.currentEpoch,
            prompts: job.prompts,
            seeds: job.seeds,
          }),
          { status: 200 },
        )
      }

      // Three-Tier Progressive Escalation Gate
      if (
        request.method === 'POST' &&
        url.pathname.endsWith('/tasks/triage-and-remediate')
      ) {
        const body = (await request.json().catch(() => ({}))) as any
        const { candidate, vlmRawOutput, workflowAst } = body as {
          candidate: CandidatePlate
          vlmRawOutput?: string
          workflowAst?: ComfyWorkflow
        }

        if (!this.activeJob || this.activeJob.taskId !== candidate.taskId) {
          return new Response(JSON.stringify({ error: 'TASK_NOT_ACTIVE' }), {
            status: 404,
          })
        }

        const taskHistory = this.auditHistory.get(candidate.taskId) || []
        const gauntletResult = await this.driftGauntlet.evaluate(
          candidate,
          taskHistory,
          vlmRawOutput ? async () => vlmRawOutput : undefined,
        )

        // Handle Gauntlet Pass
        if (gauntletResult.passed) {
          await this.ctx.storage.deleteAlarm()
          const canonicalKey = `canonical/renders/${this.activeJob.taskId}/${candidate.candidateSha}.mp4`
          this.ledgerChain = await appendLedgerEntry(
            this.ledgerChain,
            'ARTIFACT_PROMOTED',
            {
              canonicalKey,
              candidateSha: candidate.candidateSha,
            },
          )
          this.activeJob = null
          await this.persistState()

          return new Response(
            JSON.stringify({ ok: true, promoted: true, canonicalKey }),
            { status: 200 },
          )
        }

        // Handle Fatal Format Check
        if (gauntletResult.terminalStage === 'TIER_1') {
          return new Response(
            JSON.stringify({
              ok: false,
              status: 'FATAL_FORMAT_ERROR',
              diagnostic: gauntletResult.diagnostic,
            }),
            { status: 400 },
          )
        }

        // Record cycle in audit history
        if (gauntletResult.auditReport) {
          taskHistory.push({
            attempt: this.activeJob.attemptCount + 1,
            classification: gauntletResult.auditReport.classification,
            errorTokens:
              gauntletResult.auditReport.scanTrace.anachronismsDetected,
            timestampMs: Date.now(),
          })
          this.auditHistory.set(candidate.taskId, taskHistory)
        }

        const headHash =
          this.ledgerChain[this.ledgerChain.length - 1]?.blockHash ||
          '0'.repeat(64)

        // Progressive Escalation Policy Evaluation
        const directive: EscalationDirective =
          this.escalationEngine.evaluateEscalation({
            taskId: this.activeJob.taskId,
            candidateSha: candidate.candidateSha,
            attemptCount: this.activeJob.attemptCount + 1,
            maxAttempts: this.activeJob.maxAttempts,
            trajectory:
              gauntletResult.circuitBreaker?.trajectory || 'CONVERGENCE',
            auditHistory: taskHistory,
            latestReport: gauntletResult.auditReport!,
            currentJob: this.activeJob,
            currentWorkflow: workflowAst,
            ledgerHeadHash: headHash,
          })

        if (
          directive.action === 'RETRY_MICRO_PATCH' ||
          directive.action === 'RETRY_MACRO_AST'
        ) {
          this.currentEpoch += 1
          this.fairQueue.requeue(directive.remediatedJob!)
          this.ledgerChain = await appendLedgerEntry(
            this.ledgerChain,
            'ESCALATION_TIER_ENGAGED',
            {
              taskId: this.activeJob.taskId,
              tier: directive.tier,
              epoch: this.currentEpoch,
              attemptCount: directive.remediatedJob!.attemptCount,
              rationale: directive.rationale,
            },
          )

          this.activeJob = null
          await this.ctx.storage.deleteAlarm()
          await this.persistState()

          return new Response(
            JSON.stringify({
              ok: true,
              status: 'RETRY_SCHEDULED',
              tier: directive.tier,
              diagnostic: directive.rationale,
            }),
            { status: 200 },
          )
        }

        // TIER 3: Story Architect Quarantine & Compute Freeze
        this.ledgerChain = await appendLedgerEntry(
          this.ledgerChain,
          'TIER_3_QUARANTINED',
          {
            taskId: this.activeJob.taskId,
            dossierId: directive.dossier?.dossierId,
            trigger: directive.dossier?.escalationTrigger,
          },
        )

        if (this.env?.CHATOPS_WEBHOOK_URL && directive.dossier) {
          const alertText = formatDossierForChatOps(directive.dossier)
          this.chatOpsGateway
            .dispatchAlert(this.env.CHATOPS_WEBHOOK_URL, alertText)
            .catch(() => {})
        }

        this.activeJob = null
        await this.ctx.storage.deleteAlarm()
        await this.persistState()

        return new Response(
          JSON.stringify({
            ok: false,
            status: 'BLOCKED',
            tier: 'TIER_3_ARCHITECT_ESCALATION',
            dossier: directive.dossier,
            diagnostic: directive.rationale,
          }),
          { status: 200 },
        )
      }

      return new Response('NOT_FOUND', { status: 404 })
    } catch (err: any) {
      console.error(`>> [DO:ERR] ${err.message}`)
      return new Response(JSON.stringify({ error: err.message }), {
        status: 500,
      })
    }
  }

  private async persistState(): Promise<void> {
    await this.ctx.storage.put('pending_queue', this.fairQueue.toArray())
    await this.ctx.storage.put('active_job', this.activeJob)
    await this.ctx.storage.put('current_epoch', this.currentEpoch)
    await this.ctx.storage.put('ledger_chain', this.ledgerChain)
    await this.ctx.storage.put(
      'audit_history',
      Object.fromEntries(this.auditHistory),
    )
  }
}
