#### 2. Staleness Math & Health Scoring Engine (`apps/worker/src/fleet/stalenessMath.ts`)

- **Staleness State & Rating:**
- `HEALTHY`: $H \ge 0.85$ (heartbeat fresh, low jitter).
- `DEGRADED`: $0.50 \le H < 0.85$ (mild heartbeat lag or high VRAM utilization).
- `STALE`: $0.0 < H < 0.50$ (severe heartbeat delay, eligible for lease
  revocation).
- `DEAD_ZOMBIE`: $H = 0.0$ ($\Delta t \ge 60\,000\text{ ms}$, immediate
  eviction).

- **Mathematical Methods:**
- `calculateTimeStaleness(lastHeartbeatMs: number, nowMs?: number): { deltaMs: number; healthScore: number; rating: RunnerHealthRating }`:
- Calculates $\Delta t = \max(0, T_{\text{now}} - T_{\text{heartbeat}})$.
- Applies jitter buffer $\epsilon = 2\,500\text{ ms}$ over nominal
  $\tau = 10\,000\text{ ms}$.
- Calculates exponential decay scalar $H \in [0.0, 1.0]$.
- Assigns health category rating.

- `calculateCompositeHealth(timeHealth: number, vramFreeMb: number, requiredVramMb: number): { compositeScore: number; isEligible: boolean }`:
- Multiplies temporal health by memory headroom factor:

$$C = H(\Delta t) \times \min\left(1.0, \frac{\text{vramFreeMb}}{\text{requiredVramMb}}\right)$$

- Asserts eligibility when $C \ge 0.50$ and
  $\text{vramFreeMb} \ge \text{requiredVramMb}$.

export interface InterrogationReport {
  workerId: string
  reachable: boolean
  latencyMs: number
  vramFreeMb: number
  vramTotalMb: number
  activeNode: string
  queueDepth: number
  error?: string
  probedAtMs: number
}

export interface ComfySystemStatsPayload {
  system?: {
    vram_free?: number
    vram_total?: number
    devices?: Array<{
      vram_free?: number
      vram_total?: number
      name?: string
    }>
  }
  exec_info?: {
    queue_remaining?: number
  }
  active_node?: string
}

export const DEFAULT_PROBE_TIMEOUT_MS = 3000

export class RemoteRunnerInterrogator {
  /**
   * Interrogates remote GPU runner via ComfyUI / system daemon endpoint.
   * Enforces strict timeout and non-blocking failure semantics.
   */
  public async interrogateRunner(
    endpointUrl: string,
    workerId: string,
    timeoutMs = DEFAULT_PROBE_TIMEOUT_MS,
  ): Promise<InterrogationReport> {
    const probedAtMs = Date.now()
    const controller = new AbortController()
    const timeoutTimer = setTimeout(() => controller.abort(), timeoutMs)

    const startTime = Date.now()
    const cleanUrl = endpointUrl.replace(/\/+$/, '')
    const targetUrl = `${cleanUrl}/system_stats`

    try {
      const response = await fetch(targetUrl, {
        method: 'GET',
        headers: {
          'Accept': 'application/json',
          'User-Agent': '000.repo-bot-interrogator',
        },
        signal: controller.signal,
      })

      clearTimeout(timeoutTimer)
      const latencyMs = Math.max(1, Date.now() - startTime)

      if (!response.ok) {
        console.warn(
          `>> [INTERROGATE:WARN] Runner '${workerId}' returned HTTP ${response.status}`,
        )
        return {
          workerId,
          reachable: false,
          latencyMs,
          vramFreeMb: 0,
          vramTotalMb: 0,
          activeNode: 'UNAVAILABLE',
          queueDepth: 0,
          error: `HTTP_${response.status}`,
          probedAtMs,
        }
      }

      const data = (await response
        .json()
        .catch(() => ({}))) as ComfySystemStatsPayload

      // Extract VRAM from devices array or fallback fields
      let vramFree = 0
      let vramTotal = 0

      if (data.system?.devices && data.system.devices.length > 0) {
        const primaryDevice = data.system.devices[0]
        vramFree = Math.round((primaryDevice.vram_free || 0) / (1024 * 1024))
        vramTotal = Math.round((primaryDevice.vram_total || 0) / (1024 * 1024))
      } else if (data.system) {
        vramFree = Math.round((data.system.vram_free || 0) / (1024 * 1024))
        vramTotal = Math.round((data.system.vram_total || 0) / (1024 * 1024))
      }

      const queueDepth = data.exec_info?.queue_remaining ?? 0
      const activeNode =
        data.active_node || (queueDepth > 0 ? 'EXECUTING' : 'IDLE')

      console.log(
        `>> [INTERROGATE:OK] Interrogated '${workerId}' (${latencyMs}ms, VRAM: ${vramFree}/${vramTotal}MB) [OK]`,
      )

      return {
        workerId,
        reachable: true,
        latencyMs,
        vramFreeMb: vramFree,
        vramTotalMb: vramTotal,
        activeNode,
        queueDepth,
        probedAtMs,
      }
    } catch (err: any) {
      clearTimeout(timeoutTimer)
      const latencyMs = Date.now() - startTime
      const isTimeout = err?.name === 'AbortError' || controller.signal.aborted
      const errorMsg = isTimeout
        ? `PROBE_TIMEOUT_${timeoutMs}MS`
        : err?.message || 'SOCKET_ERROR'

      console.warn(
        `>> [INTERROGATE:FAIL] Failed probing '${workerId}': ${errorMsg}`,
      )

      return {
        workerId,
        reachable: false,
        latencyMs,
        vramFreeMb: 0,
        vramTotalMb: 0,
        activeNode: 'UNREACHABLE',
        queueDepth: 0,
        error: errorMsg,
        probedAtMs,
      }
    }
  }
}

export type RunnerHealthRating =
  'HEALTHY' | 'DEGRADED' | 'STALE' | 'DEAD_ZOMBIE'

export interface StalenessMetrics {
  deltaMs: number
  effectiveDeltaMs: number
  healthScore: number
  rating: RunnerHealthRating
  isEligible: boolean
}

export interface CompositeHealthResult {
  timeHealth: number
  memoryHealth: number
  compositeScore: number
  isEligible: boolean
  rating: RunnerHealthRating
}

export const NOMINAL_HEARTBEAT_INTERVAL_MS = 10_000 // tau = 10s
export const JITTER_TOLERANCE_BUFFER_MS = 2_500 // epsilon = 2.5s
export const HALF_LIFE_DECAY_MS = 15_000 // T_1/2 = 15s
export const HARD_EVICTION_THRESHOLD_MS = 60_000 // T_evict = 60s
export const MIN_ALLOCATION_HEALTH_SCORE = 0.5 // Minimum health to pull tasks

/**
 * Calculates temporal staleness using exponential decay with network jitter absorption.
 */
export function calculateTimeStaleness(
  lastHeartbeatMs: number,
  nowMs: number = Date.now(),
): StalenessMetrics {
  const deltaMs = Math.max(0, nowMs - lastHeartbeatMs)
  const graceWindow = NOMINAL_HEARTBEAT_INTERVAL_MS + JITTER_TOLERANCE_BUFFER_MS

  // Hard boundary: Runner dead beyond 60 seconds
  if (deltaMs >= HARD_EVICTION_THRESHOLD_MS) {
    return {
      deltaMs,
      effectiveDeltaMs: deltaMs - graceWindow,
      healthScore: 0.0,
      rating: 'DEAD_ZOMBIE',
      isEligible: false,
    }
  }

  // Within nominal window + network jitter tolerance: perfect health
  if (deltaMs <= graceWindow) {
    return {
      deltaMs,
      effectiveDeltaMs: 0,
      healthScore: 1.0,
      rating: 'HEALTHY',
      isEligible: true,
    }
  }

  // Calculate exponential decay past jitter window
  const effectiveDeltaMs = deltaMs - graceWindow
  const lambda = Math.LN2 / HALF_LIFE_DECAY_MS
  const rawScore = Math.exp(-lambda * effectiveDeltaMs)
  const healthScore = Math.round(rawScore * 1000) / 1000

  let rating: RunnerHealthRating = 'HEALTHY'
  if (healthScore >= 0.85) {
    rating = 'HEALTHY'
  } else if (healthScore >= MIN_ALLOCATION_HEALTH_SCORE) {
    rating = 'DEGRADED'
  } else {
    rating = 'STALE'
  }

  return {
    deltaMs,
    effectiveDeltaMs,
    healthScore,
    rating,
    isEligible: healthScore >= MIN_ALLOCATION_HEALTH_SCORE,
  }
}

/**
 * Computes composite health combining temporal staleness and VRAM memory headroom.
 */
export function calculateCompositeHealth(
  timeHealth: number,
  vramFreeMb: number,
  requiredVramMb: number,
): CompositeHealthResult {
  if (timeHealth <= 0.0 || vramFreeMb <= 0) {
    return {
      timeHealth,
      memoryHealth: 0.0,
      compositeScore: 0.0,
      isEligible: false,
      rating: 'DEAD_ZOMBIE',
    }
  }

  const memoryRatio =
    requiredVramMb > 0 ? Math.min(1.0, vramFreeMb / requiredVramMb) : 1.0
  const memoryHealth = Math.round(memoryRatio * 1000) / 1000

  const rawComposite = timeHealth * memoryHealth
  const compositeScore = Math.round(rawComposite * 1000) / 1000

  const meetsVram = requiredVramMb <= 0 || vramFreeMb >= requiredVramMb
  const isEligible = compositeScore >= MIN_ALLOCATION_HEALTH_SCORE && meetsVram

  let rating: RunnerHealthRating = 'HEALTHY'
  if (compositeScore >= 0.85 && meetsVram) {
    rating = 'HEALTHY'
  } else if (compositeScore >= MIN_ALLOCATION_HEALTH_SCORE && meetsVram) {
    rating = 'DEGRADED'
  } else if (timeHealth > 0.0) {
    rating = 'STALE'
  } else {
    rating = 'DEAD_ZOMBIE'
  }

  return {
    timeHealth,
    memoryHealth,
    compositeScore,
    isEligible,
    rating,
  }
}

import { FairQueue, type ExecutionJob } from '../queue/fairQueue.js'
import {
  createGenesisBlock,
  appendLedgerEntry,
  type LedgerEntry,
} from '../ledger/cryptoLedger.js'
import { formatDlqAlert, ChatOpsGateway } from '../telemetry/chatOpsGateway.js'
import { D1StateLedger } from '../db/d1Ledger.js'
import { FleetRegistry } from '../fleet/fleetRegistry.js'
import { RemoteRunnerInterrogator } from '../fleet/interrogator.js'
import {
  calculateTimeStaleness,
  calculateCompositeHealth,
  type RunnerHealthRating,
} from '../fleet/stalenessMath.js'

export class ShotCoordinatorDO {
  private fairQueue = new FairQueue()
  private activeJob: ExecutionJob | null = null
  private currentEpoch = 0
  private lastArtistId: string | null = null
  private ledgerChain: LedgerEntry[] = []
  private d1Ledger?: D1StateLedger
  private fleetRegistry?: FleetRegistry
  private interrogator = new RemoteRunnerInterrogator()
  private chatOpsGateway = new ChatOpsGateway()

  // In-memory worker heartbeat tracking: workerId -> lastHeartbeatMs
  private workerHeartbeats = new Map<string, number>()

  constructor(
    private ctx: any,
    private env: any,
  ) {
    if (this.env?.DB) {
      this.d1Ledger = new D1StateLedger(this.env.DB)
      this.fleetRegistry = new FleetRegistry(this.env.DB)
    }

    this.ctx.blockConcurrencyWhile?.(async () => {
      const queueData = await this.ctx.storage.get('pending_queue')
      if (Array.isArray(queueData)) this.fairQueue = new FairQueue(queueData)

      const active = await this.ctx.storage.get('active_job')
      if (active) this.activeJob = active

      const epoch = await this.ctx.storage.get('current_epoch')
      if (typeof epoch === 'number') this.currentEpoch = epoch

      const chain = await this.ctx.storage.get('ledger_chain')
      if (Array.isArray(chain)) this.ledgerChain = chain

      const heartbeats = await this.ctx.storage.get('worker_heartbeats')
      if (heartbeats && typeof heartbeats === 'object') {
        this.workerHeartbeats = new Map(Object.entries(heartbeats))
      }

      console.log(
        `>> [DO:BOOT] ShotCoordinatorDO rehydrated. Epoch: ${this.currentEpoch}, Heartbeats: ${this.workerHeartbeats.size} [OK]`,
      )
    })
  }

  private mirrorToD1(fn: (ledger: D1StateLedger) => Promise<void>): void {
    if (!this.d1Ledger) return
    const p = fn(this.d1Ledger).catch((err) => {
      console.error(
        `>> [D1:SYNC_ERR] Failed asynchronous D1 mirror: ${err.message} [FAIL]`,
      )
    })

    if (this.ctx.waitUntil) {
      this.ctx.waitUntil(p)
    }
  }

  async alarm(): Promise<void> {
    console.warn(
      `>> [WATCHDOG:TRIP] Lease timeout for task '${this.activeJob?.taskId || 'NONE'}' [ALERT]`,
    )
    if (!this.activeJob) {
      await this.ctx.storage.deleteAlarm()
      return
    }

    const task = this.activeJob
    const previousEpoch = this.currentEpoch
    task.attemptCount += 1
    this.currentEpoch += 1

    if (task.attemptCount < task.maxAttempts) {
      this.fairQueue.requeue(task)
      this.activeJob = null
      this.ledgerChain = await appendLedgerEntry(
        this.ledgerChain,
        'WATCHDOG_TIMEOUT_REQUEUE',
        {
          taskId: task.taskId,
          attempt: task.attemptCount,
          newEpoch: this.currentEpoch,
        },
      )

      this.mirrorToD1(async (ledger) => {
        await ledger.recordStateTransition({
          taskId: task.taskId,
          fromState: 'RUNNING',
          toState: 'RETRYING',
          epoch: previousEpoch,
          timestampMs: Date.now(),
          metadata: { attempt: task.attemptCount },
        })
        await ledger.recordTaskSnapshot({
          taskId: task.taskId,
          status: 'RETRYING',
          currentEpoch: this.currentEpoch,
          artistId: task.artistId,
          workflowTemplate: task.workflowTemplate,
          attemptCount: task.attemptCount,
          createdAtMs: task.enqueuedAt,
          updatedAtMs: Date.now(),
        })
      })
    } else {
      this.activeJob = null
      this.ledgerChain = await appendLedgerEntry(
        this.ledgerChain,
        'CIRCUIT_BREAKER_DLQ',
        {
          taskId: task.taskId,
          attemptCount: task.attemptCount,
        },
      )

      this.mirrorToD1(async (ledger) => {
        await ledger.recordStateTransition({
          taskId: task.taskId,
          fromState: 'RUNNING',
          toState: 'DLQ',
          epoch: previousEpoch,
          timestampMs: Date.now(),
          metadata: { reason: 'MAX_ATTEMPTS_EXHAUSTED' },
        })
        await ledger.recordTaskSnapshot({
          taskId: task.taskId,
          status: 'DLQ',
          currentEpoch: this.currentEpoch,
          artistId: task.artistId,
          workflowTemplate: task.workflowTemplate,
          attemptCount: task.attemptCount,
          createdAtMs: task.enqueuedAt,
          updatedAtMs: Date.now(),
        })
      })

      if (this.env?.CHATOPS_WEBHOOK_URL) {
        const dlqAlert = formatDlqAlert(
          task.taskId,
          task.attemptCount,
          'MAX_ATTEMPTS_EXHAUSTED',
        )
        this.chatOpsGateway
          .dispatchAlert(this.env.CHATOPS_WEBHOOK_URL, dlqAlert)
          .catch(() => {})
      }
    }

    await this.ctx.storage.deleteAlarm()
    await this.persistState()
  }

  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url)

    try {
      // Task Enqueue Route
      if (request.method === 'POST' && url.pathname.endsWith('/enqueue')) {
        const body = (await request.json().catch(() => ({}))) as any
        const {
          taskId,
          artistId,
          idempotencyKey,
          workflowTemplate = 'default.json',
          prompts = {},
          seeds = [42],
          maxAttempts = 3,
          requiredVramMb = 0,
        } = body

        const job: ExecutionJob = {
          taskId,
          artistId,
          idempotencyKey,
          workflowTemplate,
          prompts: { ...prompts, requiredVramMb },
          seeds,
          enqueuedAt: Date.now(),
          attemptCount: 0,
          maxAttempts,
        }

        const enqueued = this.fairQueue.enqueue(job)
        if (!enqueued) {
          return new Response(JSON.stringify({ error: 'DUPLICATE_TASK' }), {
            status: 409,
          })
        }

        if (this.ledgerChain.length === 0) {
          this.ledgerChain = [
            await createGenesisBlock(taskId, { enqueuedAt: Date.now() }),
          ]
        } else {
          this.ledgerChain = await appendLedgerEntry(
            this.ledgerChain,
            'TASK_ENQUEUED',
            { taskId, artistId },
          )
        }

        this.mirrorToD1(async (ledger) => {
          await ledger.recordTaskSnapshot({
            taskId: job.taskId,
            status: 'PENDING',
            currentEpoch: this.currentEpoch,
            artistId: job.artistId,
            workflowTemplate: job.workflowTemplate,
            attemptCount: 0,
            createdAtMs: job.enqueuedAt,
            updatedAtMs: Date.now(),
          })
          await ledger.recordStateTransition({
            taskId: job.taskId,
            fromState: 'NONE',
            toState: 'PENDING',
            epoch: this.currentEpoch,
            timestampMs: Date.now(),
          })
        })

        await this.persistState()
        return new Response(
          JSON.stringify({ ok: true, taskId, position: this.fairQueue.size() }),
          { status: 200 },
        )
      }

      // Lease Heartbeat Ingress
      if (
        request.method === 'POST' &&
        url.pathname.endsWith('/leases/heartbeat')
      ) {
        const body = (await request.json().catch(() => ({}))) as any
        const { workerId, epoch, vramFreeMb } = body

        if (epoch !== this.currentEpoch) {
          return new Response(
            JSON.stringify({
              error: 'STALE_EPOCH_CONFLICT',
              activeEpoch: this.currentEpoch,
            }),
            {
              status: 409,
            },
          )
        }

        this.workerHeartbeats.set(workerId, Date.now())
        await this.ctx.storage.put(
          'worker_heartbeats',
          Object.fromEntries(this.workerHeartbeats),
        )

        if (this.fleetRegistry) {
          this.ctx.waitUntil?.(
            this.fleetRegistry.recordHeartbeat(
              workerId,
              vramFreeMb || 0,
              this.activeJob?.taskId,
            ),
          )
        }

        return new Response(
          JSON.stringify({ ok: true, epoch: this.currentEpoch }),
          { status: 200 },
        )
      }

      // Lease Claim Route with Staleness & Health Gating
      if (request.method === 'POST' && url.pathname.endsWith('/leases/claim')) {
        const body = (await request.json().catch(() => ({}))) as any
        const workerId = body.workerId || 'rig2_gpu'
        const endpointUrl = body.endpointUrl

        if (this.activeJob !== null) {
          return new Response(JSON.stringify({ ok: false, busy: true }), {
            status: 200,
          })
        }

        // Staleness check on requesting worker
        const lastHeartbeat = this.workerHeartbeats.get(workerId)
        if (lastHeartbeat) {
          const staleness = calculateTimeStaleness(lastHeartbeat)
          if (!staleness.isEligible) {
            return new Response(
              JSON.stringify({
                error: 'WORKER_STALE_GATED',
                healthScore: staleness.healthScore,
                rating: staleness.rating,
              }),
              { status: 409 },
            )
          }
        }

        const job = this.fairQueue.dequeueNextFair(this.lastArtistId)
        if (!job) return new Response(null, { status: 204 })

        // Capability and remote interrogation check if endpoint supplied
        if (endpointUrl) {
          const probe = await this.interrogator.interrogateRunner(
            endpointUrl,
            workerId,
            2000,
          )
          const requiredVram = job.prompts?.requiredVramMb || 0
          const composite = calculateCompositeHealth(
            1.0,
            probe.vramFreeMb,
            requiredVram,
          )

          if (!probe.reachable || !composite.isEligible) {
            // Re-queue task and reject claim
            this.fairQueue.requeue(job)
            return new Response(
              JSON.stringify({
                error: 'INTERROGATION_FAILED',
                reachable: probe.reachable,
                compositeScore: composite.compositeScore,
                vramFreeMb: probe.vramFreeMb,
                requiredVramMb: requiredVram,
              }),
              { status: 503 },
            )
          }
        }

        this.currentEpoch += 1
        this.activeJob = job
        this.lastArtistId = job.artistId
        this.workerHeartbeats.set(workerId, Date.now())

        this.ledgerChain = await appendLedgerEntry(
          this.ledgerChain,
          'LEASE_CLAIMED',
          {
            taskId: job.taskId,
            epoch: this.currentEpoch,
            workerId,
          },
        )

        this.mirrorToD1(async (ledger) => {
          await ledger.recordStateTransition({
            taskId: job.taskId,
            fromState: 'PENDING',
            toState: 'RUNNING',
            epoch: this.currentEpoch,
            workerId,
            timestampMs: Date.now(),
          })
          await ledger.recordTaskSnapshot({
            taskId: job.taskId,
            status: 'RUNNING',
            currentEpoch: this.currentEpoch,
            artistId: job.artistId,
            workflowTemplate: job.workflowTemplate,
            attemptCount: job.attemptCount,
            createdAtMs: job.enqueuedAt,
            updatedAtMs: Date.now(),
          })
        })

        await this.ctx.storage.setAlarm(Date.now() + 30_000)
        await this.persistState()

        return new Response(
          JSON.stringify({
            ok: true,
            taskId: job.taskId,
            epoch: this.currentEpoch,
            prompts: job.prompts,
            seeds: job.seeds,
          }),
          { status: 200 },
        )
      }

      // Fleet Health Inspection Route
      if (request.method === 'GET' && url.pathname.endsWith('/fleet/health')) {
        const now = Date.now()
        const healthReport: Record<string, any> = {}

        for (const [workerId, lastHb] of this.workerHeartbeats.entries()) {
          const staleness = calculateTimeStaleness(lastHb, now)
          healthReport[workerId] = {
            lastHeartbeatMs: lastHb,
            deltaMs: staleness.deltaMs,
            healthScore: staleness.healthScore,
            rating: staleness.rating,
            isEligible: staleness.isEligible,
          }
        }

        return new Response(
          JSON.stringify({
            ok: true,
            activeEpoch: this.currentEpoch,
            fleet: healthReport,
          }),
          {
            status: 200,
            headers: { 'Content-Type': 'application/json' },
          },
        )
      }

      return new Response('NOT_FOUND', { status: 404 })
    } catch (err: any) {
      console.error(`>> [DO:ERR] ${err.message}`)
      return new Response(JSON.stringify({ error: err.message }), {
        status: 500,
      })
    }
  }

  private async persistState(): Promise<void> {
    await this.ctx.storage.put('pending_queue', this.fairQueue.toArray())
    await this.ctx.storage.put('active_job', this.activeJob)
    await this.ctx.storage.put('current_epoch', this.currentEpoch)
    await this.ctx.storage.put('ledger_chain', this.ledgerChain)
    await this.ctx.storage.put(
      'worker_heartbeats',
      Object.fromEntries(this.workerHeartbeats),
    )
  }
}
