#### 1. Cryptographic Hash-Chain Ledger Engine (`apps/worker/src/ledger/cryptoLedger.ts`)

- **Ledger Block Model (`LedgerEntry`):**
- `index`: Monotonically increasing sequence integer ($0, 1, 2, \dots$).
- `previousHash`: Hex-encoded SHA-256 hash of block $k-1$ (Genesis block points
  to `'0'.repeat(64)`).
- `timestampMs`: Edge millisecond timestamp.
- `eventType`: Deterministic event identifier (e.g., `'TASK_ENQUEUED'`,
  `'LEASE_CLAIMED'`, `'WATCHDOG_TRIPPED'`, `'ARTIFACT_PROMOTED'`,
  `'DLQ_QUARANTINED'`).
- `payload`: Structured event payload.
- `blockHash`: Canonical SHA-256 digest of serialized block data.

- **Ledger Utilities:**
- `sha256Hex(data: string): Promise<string>`: Computes SHA-256 hex string using
  native `crypto.subtle.digest`.
- `createGenesisBlock(taskId: string, initialPayload?: Record<string, any>): Promise<LedgerEntry>`:
  Seeds index 0 with fixed previous hash.
- `appendLedgerEntry(chain: LedgerEntry[], eventType: string, payload: Record<string, any>): Promise<LedgerEntry[]>`:
  Computes $H_k$, verifies continuity, appends new block, and returns updated
  chain.
- `verifyLedgerIntegrity(chain: LedgerEntry[]): Promise<{ valid: boolean; brokenIndex?: number; error?: string }>`:
  Iterates sequentially, recalculates digests, and verifies previous hash links.

export interface LedgerEntry {
  index: number
  previousHash: string
  timestampMs: number
  eventType: string
  payload: Record<string, any>
  blockHash: string
}

export interface LedgerAuditResult {
  valid: boolean
  brokenIndex?: number
  error?: string
}

export const GENESIS_PREVIOUS_HASH = '0'.repeat(64)

/**
 * Computes a standard SHA-256 hex digest using Web Crypto API.
 */
export async function sha256Hex(data: string): Promise<string> {
  const encoder = new TextEncoder()
  const buffer = encoder.encode(data)
  const hashBuffer = await crypto.subtle.digest('SHA-256', buffer)
  const hashArray = Array.from(new Uint8Array(hashBuffer))
  return hashArray.map((b) => b.toString(16).padStart(2, '0')).join('')
}

/**
 * Serializes block data into canonical deterministic string representation.
 */
export function serializeBlockData(
  index: number,
  previousHash: string,
  timestampMs: number,
  eventType: string,
  payload: Record<string, any>,
): string {
  return JSON.stringify({
    index,
    previousHash,
    timestampMs,
    eventType,
    payload,
  })
}

/**
 * Creates and hashes the initial Genesis block for a task ledger.
 */
export async function createGenesisBlock(
  taskId: string,
  initialPayload: Record<string, any> = {},
): Promise<LedgerEntry> {
  const index = 0
  const previousHash = GENESIS_PREVIOUS_HASH
  const timestampMs = Date.now()
  const eventType = 'TASK_GENESIS'
  const payload = { taskId, ...initialPayload }

  const serialized = serializeBlockData(
    index,
    previousHash,
    timestampMs,
    eventType,
    payload,
  )
  const blockHash = await sha256Hex(serialized)

  return {
    index,
    previousHash,
    timestampMs,
    eventType,
    payload,
    blockHash,
  }
}

/**
 * Appends a new verified block to the hash chain.
 */
export async function appendLedgerEntry(
  chain: LedgerEntry[],
  eventType: string,
  payload: Record<string, any>,
  timestampMs: number = Date.now(),
): Promise<LedgerEntry[]> {
  if (chain.length === 0) {
    throw new Error(
      'LEDGER_INVALID: Cannot append to an empty chain. Genesis block required.',
    )
  }

  const previousBlock = chain[chain.length - 1]
  const index = previousBlock.index + 1
  const previousHash = previousBlock.blockHash

  const serialized = serializeBlockData(
    index,
    previousHash,
    timestampMs,
    eventType,
    payload,
  )
  const blockHash = await sha256Hex(serialized)

  const newEntry: LedgerEntry = {
    index,
    previousHash,
    timestampMs,
    eventType,
    payload,
    blockHash,
  }

  console.log(
    `>> [LEDGER:APPEND] Appended block ${index} (${eventType}) Hash: ${blockHash.substring(0, 12)}... [OK]`,
  )
  return [...chain, newEntry]
}

/**
 * Verifies the mathematical integrity of the entire cryptographic hash chain.
 */
export async function verifyLedgerIntegrity(
  chain: LedgerEntry[],
): Promise<LedgerAuditResult> {
  if (chain.length === 0) {
    return { valid: false, error: 'CHAIN_EMPTY' }
  }

  // Verify Genesis block
  const genesis = chain[0]
  if (genesis.index !== 0 || genesis.previousHash !== GENESIS_PREVIOUS_HASH) {
    return { valid: false, brokenIndex: 0, error: 'INVALID_GENESIS_POINTER' }
  }

  const expectedGenesisHash = await sha256Hex(
    serializeBlockData(
      genesis.index,
      genesis.previousHash,
      genesis.timestampMs,
      genesis.eventType,
      genesis.payload,
    ),
  )

  if (genesis.blockHash !== expectedGenesisHash) {
    return { valid: false, brokenIndex: 0, error: 'GENESIS_HASH_MISMATCH' }
  }

  // Sequentially verify downstream links
  for (let i = 1; i < chain.length; i++) {
    const prev = chain[i - 1]
    const current = chain[i]

    // 1. Index continuity
    if (current.index !== prev.index + 1) {
      return {
        valid: false,
        brokenIndex: i,
        error: `INDEX_DISCONTINUITY: Expected ${prev.index + 1}, found ${current.index}`,
      }
    }

    // 2. Previous hash pointer continuity
    if (current.previousHash !== prev.blockHash) {
      return {
        valid: false,
        brokenIndex: i,
        error: `PREVIOUS_HASH_MISMATCH at block ${i}`,
      }
    }

    // 3. Current block hash integrity
    const calculatedHash = await sha256Hex(
      serializeBlockData(
        current.index,
        current.previousHash,
        current.timestampMs,
        current.eventType,
        current.payload,
      ),
    )

    if (current.blockHash !== calculatedHash) {
      return {
        valid: false,
        brokenIndex: i,
        error: `BLOCK_HASH_TAMPERED at block ${i}`,
      }
    }
  }

  console.log(
    `>> [LEDGER:AUDIT] Verified unbroken cryptographic integrity across ${chain.length} blocks [OK]`,
  )
  return { valid: true }
}

export interface ChatOpsVerificationResult {
  valid: boolean
  error?: string
}

export const MAX_DRIFT_WINDOW_MS = 300_000 // 5 minutes anti-replay window

/**
 * Converts a hex string into a Uint8Array byte buffer.
 */
function hexToBytes(hex: string): Uint8Array {
  const bytes = new Uint8Array(hex.length / 2)
  for (let i = 0; i < hex.length; i += 2) {
    bytes[i / 2] = parseInt(hex.substring(i, i + 2), 16)
  }
  return bytes
}

/**
 * Computes timing-safe HMAC-SHA256 signature verification using Web Crypto API.
 */
export async function verifyChatOpsSignature(
  rawBody: string,
  signatureHeader: string,
  timestampHeader: string,
  secret: string,
): Promise<ChatOpsVerificationResult> {
  const timestampSec = parseInt(timestampHeader, 10)
  if (isNaN(timestampSec)) {
    return { valid: false, error: 'INVALID_TIMESTAMP_HEADER' }
  }

  const nowSec = Math.floor(Date.now() / 1000)
  const skewSec = Math.abs(nowSec - timestampSec)

  // Anti-replay attack timing window: 300 seconds (5 minutes)
  if (skewSec > MAX_DRIFT_WINDOW_MS / 1000) {
    return {
      valid: false,
      error: `TIMESTAMP_DRIFT_EXPIRED: Skew is ${skewSec}s (limit 300s)`,
    }
  }

  // Expected format: v0=a8f5b3...
  const prefix = 'v0='
  if (!signatureHeader.startsWith(prefix)) {
    return { valid: false, error: 'INVALID_SIGNATURE_FORMAT' }
  }
  const signatureHex = signatureHeader.substring(prefix.length)

  // Message payload to sign: v0:<timestamp>:<rawBody>
  const signaturePayload = `v0:${timestampHeader}:${rawBody}`
  const encoder = new TextEncoder()
  const dataBytes = encoder.encode(signaturePayload)
  const secretBytes = encoder.encode(secret)

  try {
    const key = await crypto.subtle.importKey(
      'raw',
      secretBytes,
      { name: 'HMAC', hash: 'SHA-256' },
      false,
      ['verify', 'sign'],
    )

    const expectedSigBytes = hexToBytes(signatureHex)
    const isValid = await crypto.subtle.verify(
      'HMAC',
      key,
      expectedSigBytes,
      dataBytes,
    )

    if (!isValid) {
      return { valid: false, error: 'SIGNATURE_MISMATCH' }
    }

    return { valid: true }
  } catch (err: any) {
    return {
      valid: false,
      error: `HMAC_VERIFICATION_EXCEPTION: ${err.message}`,
    }
  }
}

/**
 * Generates an HMAC-SHA256 signature header for outbound dispatch or test simulation.
 */
export async function generateChatOpsSignature(
  rawBody: string,
  timestampSec: number,
  secret: string,
): Promise<string> {
  const signaturePayload = `v0:${timestampSec}:${rawBody}`
  const encoder = new TextEncoder()
  const key = await crypto.subtle.importKey(
    'raw',
    encoder.encode(secret),
    { name: 'HMAC', hash: 'SHA-256' },
    false,
    ['sign'],
  )

  const sigBuffer = await crypto.subtle.sign(
    'HMAC',
    key,
    encoder.encode(signaturePayload),
  )
  const sigHex = Array.from(new Uint8Array(sigBuffer))
    .map((b) => b.toString(16).padStart(2, '0'))
    .join('')

  return `v0=${sigHex}`
}

import {
  verifyChatOpsSignature,

export interface TaskRecord {
  taskId: string
  fsmState: string
  currentEpoch: number
  attemptCount: number
  maxAttempts: number
  canonicalManifest: Record<string, any> | null
  updatedAt: number
}

export class ShotCoordinatorDO {
  private taskId = ''
  private fsmState = 'PENDING'
  private currentEpoch = 0
  private attemptCount = 0
  private maxAttempts = 3
  private canonicalManifest: Record<string, any> | null = null
  private ledgerChain: LedgerEntry[] = []
  private chatOpsGateway = new ChatOpsGateway()

  constructor(
    private ctx: any,
    private env: any,
  ) {
    this.ctx.blockConcurrencyWhile?.(async () => {
      const storedRecord = await this.ctx.storage.get('task_record')
      if (storedRecord && typeof storedRecord === 'object') {
        this.taskId = storedRecord.taskId || ''
        this.fsmState = storedRecord.fsmState || 'PENDING'
        this.currentEpoch = storedRecord.currentEpoch || 0
        this.attemptCount = storedRecord.attemptCount || 0
        this.maxAttempts = storedRecord.maxAttempts || 3
        this.canonicalManifest = storedRecord.canonicalManifest || null
      }

      const storedChain = await this.ctx.storage.get('ledger_chain')
      if (Array.isArray(storedChain)) {
        this.ledgerChain = storedChain
      }

      console.log(
        `>> [DO:BOOT] ShotCoordinatorDO loaded. Task '${this.taskId}' Epoch: ${this.currentEpoch}, Ledger Blocks: ${this.ledgerChain.length} [OK]`,
      )
    })
  }

  /**
   * Hardware Watchdog Alarm Handler.
   */
  async alarm(): Promise<void> {
    console.warn(
      `>> [WATCHDOG:ALARM] Watchdog alarm fired for task '${this.taskId}' [ALERT]`,
    )

    this.attemptCount += 1
    this.currentEpoch += 1

    if (this.attemptCount < this.maxAttempts) {
      this.fsmState = 'RETRYING'
      this.ledgerChain = await appendLedgerEntry(
        this.ledgerChain,
        'WATCHDOG_TIMEOUT_RETRY',
        {
          attempt: this.attemptCount,
          newEpoch: this.currentEpoch,
        },
      )

      // Asynchronous ChatOps Alert
      const alertText = formatWatchdogAlert(
        this.taskId,
        this.attemptCount,
        this.maxAttempts,
        this.currentEpoch,
      )
      if (this.env?.CHATOPS_WEBHOOK_URL) {
        this.chatOpsGateway
          .dispatchAlert(this.env.CHATOPS_WEBHOOK_URL, alertText)
          .catch(() => {})
      }
    } else {
      this.fsmState = 'DLQ'
      this.ledgerChain = await appendLedgerEntry(
        this.ledgerChain,
        'CIRCUIT_BREAKER_DLQ',
        {
          attemptCount: this.attemptCount,
          error: 'MAX_ATTEMPTS_EXHAUSTED',
        },
      )

      const dlqText = formatDlqAlert(
        this.taskId,
        this.attemptCount,
        'MAX_ATTEMPTS_EXHAUSTED',
      )
      if (this.env?.CHATOPS_WEBHOOK_URL) {
        this.chatOpsGateway
          .dispatchAlert(this.env.CHATOPS_WEBHOOK_URL, dlqText)
          .catch(() => {})
      }
    }

    await this.ctx.storage.deleteAlarm()
    await this.persist()
  }

  /**
   * HTTP / RPC Interface.
   */
  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url)

    try {
      // Task Enqueue Route
      if (request.method === 'POST' && url.pathname.endsWith('/enqueue')) {
        const body = (await request.json().catch(() => ({}))) as any
        this.taskId = body.taskId || 'task-default'
        this.fsmState = 'PENDING'
        this.currentEpoch = 0
        this.attemptCount = 0

        // Initialize cryptographic genesis block
        const genesis = await createGenesisBlock(this.taskId, {
          enqueuedAt: Date.now(),
          workflowTemplate: body.workflowTemplate || 'default.json',
        })
        this.ledgerChain = [genesis]

        await this.persist()

        return new Response(
          JSON.stringify({
            ok: true,
            taskId: this.taskId,
            genesisHash: genesis.blockHash,
          }),
          {
            status: 200,
            headers: { 'Content-Type': 'application/json' },
          },
        )
      }

      // Lease Claim Route
      if (request.method === 'POST' && url.pathname.endsWith('/leases/claim')) {
        const body = (await request.json().catch(() => ({}))) as any
        const workerId = body.workerId || 'rig2_gpu'

        if (this.fsmState !== 'PENDING' && this.fsmState !== 'RETRYING') {
          return new Response(
            JSON.stringify({
              error: `TASK_UNAVAILABLE: State is '${this.fsmState}'`,
            }),
            {
              status: 409,
              headers: { 'Content-Type': 'application/json' },
            },
          )
        }

        this.currentEpoch += 1
        this.fsmState = 'RUNNING'

        this.ledgerChain = await appendLedgerEntry(
          this.ledgerChain,
          'LEASE_CLAIMED',
          {
            workerId,
            epoch: this.currentEpoch,
          },
        )

        await this.ctx.storage.setAlarm(Date.now() + 30_000)
        await this.persist()

        return new Response(
          JSON.stringify({
            ok: true,
            taskId: this.taskId,
            epoch: this.currentEpoch,
          }),
          {
            status: 200,
            headers: { 'Content-Type': 'application/json' },
          },
        )
      }

      // Task Complete Route
      if (
        request.method === 'POST' &&
        url.pathname.endsWith('/tasks/complete')
      ) {
        const body = (await request.json().catch(() => ({}))) as any
        const { epoch, stagingKey, sha256 } = body

        // Monotonic Epoch Verification
        if (epoch !== this.currentEpoch) {
          return new Response(
            JSON.stringify({
              error: `STALE_EPOCH_ZOMBIE: Presented ${epoch}, active is ${this.currentEpoch}`,
            }),
            { status: 409, headers: { 'Content-Type': 'application/json' } },
          )
        }

        const canonicalKey = `canonical/renders/${this.taskId}/${sha256}.mp4`
        this.canonicalManifest = {
          taskId: this.taskId,
          epoch: this.currentEpoch,
          stagingKey,
          canonicalKey,
          sha256,
          promotedAtMs: Date.now(),
        }

        this.fsmState = 'VERIFYING'
        await this.ctx.storage.deleteAlarm()

        // Append tamper-evident promotion block
        this.ledgerChain = await appendLedgerEntry(
          this.ledgerChain,
          'ARTIFACT_PROMOTED',
          {
            canonicalKey,
            sha256,
            epoch: this.currentEpoch,
          },
        )

        await this.persist()

        // Outbound ChatOps Alert
        const promoAlert = formatPromotionNotification(
          this.taskId,
          this.currentEpoch,
          canonicalKey,
          sha256,
        )
        if (this.env?.CHATOPS_WEBHOOK_URL) {
          this.chatOpsGateway
            .dispatchAlert(this.env.CHATOPS_WEBHOOK_URL, promoAlert)
            .catch(() => {})
        }

        return new Response(
          JSON.stringify({ ok: true, promoted: true, canonicalKey }),
          {
            status: 200,
            headers: { 'Content-Type': 'application/json' },
          },
        )
      }

      // Ledger Read Route
      if (request.method === 'GET' && url.pathname.endsWith('/ledger')) {
        return new Response(JSON.stringify({ chain: this.ledgerChain }), {
          status: 200,
          headers: { 'Content-Type': 'application/json' },
        })
      }

      // Ledger Integrity Audit Route
      if (request.method === 'GET' && url.pathname.endsWith('/ledger/verify')) {
        const audit = await verifyLedgerIntegrity(this.ledgerChain)
        return new Response(JSON.stringify(audit), {
          status: 200,
          headers: { 'Content-Type': 'application/json' },
        })
      }

      // Inbound ChatOps Webhook Gateway Route
      if (
        request.method === 'POST' &&
        url.pathname.endsWith('/chatops/command')
      ) {
        const sigHeader = request.headers.get('x-chatops-signature') || ''
        const timeHeader = request.headers.get('x-chatops-timestamp') || ''
        const rawBody = await request.text()
        const secret = this.env?.CHATOPS_SECRET || 'test_secret_key'

        const verification = await verifyChatOpsSignature(
          rawBody,
          sigHeader,
          timeHeader,
          secret,
        )
        if (!verification.valid) {
          return new Response(JSON.stringify({ error: verification.error }), {
            status: 401,
            headers: { 'Content-Type': 'application/json' },
          })
        }

        const cmd = JSON.parse(rawBody)
        if (cmd.action === 'AUDIT_LEDGER') {
          const audit = await verifyLedgerIntegrity(this.ledgerChain)
          const latestHash =
            this.ledgerChain[this.ledgerChain.length - 1]?.blockHash || 'NONE'
          const alert = formatLedgerAuditAlert(
            this.taskId,
            this.ledgerChain.length,
            latestHash,
          )
          return new Response(JSON.stringify({ ok: true, audit, alert }), {
            status: 200,
            headers: { 'Content-Type': 'application/json' },
          })
        }

        return new Response(JSON.stringify({ ok: true, action: cmd.action }), {
          status: 200,
          headers: { 'Content-Type': 'application/json' },
        })
      }

      return new Response('NOT_FOUND', { status: 404 })
    } catch (err: any) {
      console.error(`>> [DO:ERR] ${err.message}`)
      return new Response(JSON.stringify({ error: err.message }), {
        status: 500,
        headers: { 'Content-Type': 'application/json' },
      })
    }
  }

  private async persist(): Promise<void> {
    await this.ctx.storage.put('task_record', {
      taskId: this.taskId,
      fsmState: this.fsmState,
      currentEpoch: this.currentEpoch,
      attemptCount: this.attemptCount,
      maxAttempts: this.maxAttempts,
      canonicalManifest: this.canonicalManifest,
      updatedAt: Date.now(),
    })
    await this.ctx.storage.put('ledger_chain', this.ledgerChain)
  }
}
