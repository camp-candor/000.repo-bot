### II. ARCHITECTURAL REQUIREMENTS

#### 1. Pure Dependency Injection Context (`apps/worker/src/context/isolateContext.ts`)

- **Context Structure (`IsolateContext`):**
- Holds runtime bindings (`env: Env`), per-execution lifetime hook
  (`executionCtx?: ExecutionContext`), and request-scoped metadata (`traceId`,
  `startedAtMs`).
- Exposes scoped accessor methods to retrieve database handles, GitHub API
  clients, and AI gateway bindings without referencing ambient globals.
- Maintains an isolated, per-context telemetry buffer that prevents diagnostic
  logs from leaking across overlapping requests within the same isolate.

- **Factory Function (`createIsolateContext(env, executionCtx, options)`):**
- Validates required environment bindings (`GITHUB_TOKEN`,
  `CLOUDFLARE_ACCOUNT_ID`, etc.).
- Generates a unique deterministic execution `traceId` if not supplied in
  incoming headers.
- Returns an immutable `IsolateContext` record.

#### 2. Payload Byte Sizing & OOM Tripwire Sentinel (`apps/worker/src/telemetry/oomTripwire.ts`)

- **Constants:**
- `ISOLATE_MEMORY_CEILING_BYTES = 128 * 1024 * 1024` (128 MB maximum isolate
  budget).
- `DEFAULT_OOM_TRIPWIRE_THRESHOLD_BYTES = 96 * 1024 * 1024` (96 MB safety
  threshold).
- `MAX_SINGLE_PAYLOAD_BYTES = 32 * 1024 * 1024` (32 MB maximum single payload).

- **Payload Sizing Engine (`estimateObjectSizeBytes(val: unknown)`):**
- Traverses strings (UTF-16 calculation: 2 bytes per character), numbers (8
  bytes), booleans (4 bytes), ArrayBuffers/TypedArrays (`byteLength`), and
  nested object/array structures with cycle protection (via `WeakSet`).

- **Stateful Tripwire Controller (`createMemoryTripwire(options)`):**
- Tracks cumulative allocated bytes within the current execution pipeline.
- `reserve(bytesOrObj: number | unknown): void`: Evaluates the requested
  reservation against `tripwireThresholdBytes`.
- If exceeded: Emits ASCII telemetry
  `>> [OOM:TRIPWIRE] Budget exceeded: ${used} / ${limit} bytes [HALT]`, fires an
  optional `onTripwireExceeded` callback (for `#ops-bridge` alerting), and
  throws `Error('ISOLATE_MEMORY_BUDGET_EXCEEDED: ...')`.
- `release(bytes: number): void`: Decrements the allocation counter when buffers
  are discarded.
- `reset(): void`: Resets tracking between isolated stages.

#### 3. Automated Vitest Verification Battery (`apps/worker/test/unit/isolateMemory.test.ts`)

- **Segregation Test:** Create multiple `IsolateContext` instances with

```typescript
import type { Env } from '../tools.js'

export interface ContextOptions {
  traceId?: string
  source?: string
}

export interface ScopedTelemetryEntry {
  timestampMs: number
  level: 'INFO' | 'WARN' | 'ERROR'
  tag: string
  message: string
}

export interface IsolateContext {
  readonly env: Env
  readonly executionCtx?: {
    waitUntil: (promise: Promise<any>) => void
  }
  readonly traceId: string
  readonly startedAtMs: number
  readonly source: string

  // Scoped Diagnostics Buffer
  log(tag: string, message: string): void
  warn(tag: string, message: string): void
  error(tag: string, message: string): void
  getTelemetryEntries(): readonly ScopedTelemetryEntry[]
  clearTelemetry(): void
}

/**
 * Creates an immutable, isolated execution context container with zero ambient globals.
 */
export function createIsolateContext(
  env: Env,
  executionCtx?: { waitUntil: (promise: Promise<any>) => void },
  options: ContextOptions = {},
): IsolateContext {
  if (!env) {
    throw new Error(
      'ISOLATE_CONTEXT_INIT_FAILED: Runtime env binding is required.',
    )
  }

  const startedAtMs = Date.now()
  const traceId =
    options.traceId ||
    `trc-${startedAtMs}-${Math.random().toString(36).slice(2, 8)}`
  const source = options.source || 'WORKER_DEFAULT'

  const telemetryBuffer: ScopedTelemetryEntry[] = []

  const appendEntry = (
    level: 'INFO' | 'WARN' | 'ERROR',
    tag: string,
    message: string,
  ) => {
    const entry: ScopedTelemetryEntry = {
      timestampMs: Date.now(),
      level,
      tag,
      message,
    }
    telemetryBuffer.push(entry)
    const prefix = `>> [${level}:${tag}] [${traceId}]`
    if (level === 'ERROR') {
      console.error(`${prefix} ${message}`)
    } else if (level === 'WARN') {
      console.warn(`${prefix} ${message}`)
    } else {
      console.log(`${prefix} ${message}`)
    }
  }

  return {
    env,
    executionCtx,
    traceId,
    startedAtMs,
    source,

    log(tag: string, message: string) {
      appendEntry('INFO', tag, message)
    },
    warn(tag: string, message: string) {
      appendEntry('WARN', tag, message)
    },
    error(tag: string, message: string) {
      appendEntry('ERROR', tag, message)
    },
    getTelemetryEntries() {
      return Object.freeze([...telemetryBuffer])
    },
    clearTelemetry() {
      telemetryBuffer.length = 0
    },
  }
}
```

#### 2. Create `apps/worker/src/telemetry/oomTripwire.ts`

```typescript
export const ISOLATE_MEMORY_CEILING_BYTES = 128 * 1024 * 1024 // 128 MB V8 Isolate Limit
export const DEFAULT_OOM_TRIPWIRE_THRESHOLD_BYTES = 96 * 1024 * 1024 // 96 MB Headroom Tripwire
export const MAX_SINGLE_PAYLOAD_BYTES = 32 * 1024 * 1024 // 32 MB Maximum Single Object

export interface MemoryTripwireOptions {
  ceilingBytes?: number
  tripwireThresholdBytes?: number
  maxSinglePayloadBytes?: number
  onTripwireExceeded?: (allocatedBytes: number, limitBytes: number) => void
}

export interface MemoryTripwire {
  getAllocatedBytes(): number
  reserve(bytesOrObj: number | unknown): void
  release(bytes: number): void
  reset(): void
}

/**
 * Traverses values and calculates estimated memory consumption in bytes.
 * Safely guards against circular structures via a WeakSet tracker.
 */
export function estimateObjectSizeBytes(
  val: unknown,
  seen = new WeakSet(),
): number {
  if (val === null || val === undefined) {
    return 0
  }

  if (typeof val === 'boolean') {
    return 4
  }

  if (typeof val === 'number') {
    return 8
  }

  if (typeof val === 'string') {
    return val.length * 2 // UTF-16 in V8: 2 bytes per char
  }

  if (typeof val === 'symbol') {
    return 32
  }

  if (typeof val === 'object') {
    if (seen.has(val)) {
      return 0 // Prevent infinite cycle recursion
    }
    seen.add(val)

    if (val instanceof ArrayBuffer) {
      return val.byteLength
    }

    if (ArrayBuffer.isView(val)) {
      return val.byteLength
    }

    let bytes = 32 // Base object pointer overhead

    if (Array.isArray(val)) {
      for (let i = 0; i < val.length; i++) {
        bytes += 8 + estimateObjectSizeBytes(val[i], seen)
      }
      return bytes
    }

    for (const key of Object.keys(val)) {
      bytes += key.length * 2 + 8 // Key string plus property descriptor
      bytes += estimateObjectSizeBytes((val as any)[key], seen)
    }

    return bytes
  }

  return 16
}

/**
 * Constructs a fail-closed memory allocation tripwire to protect worker isolates from OOM crashes.
 */
export function createMemoryTripwire(
  options: MemoryTripwireOptions = {},
): MemoryTripwire {
  const thresholdBytes =
    options.tripwireThresholdBytes ?? DEFAULT_OOM_TRIPWIRE_THRESHOLD_BYTES
  const maxSingleBytes =
    options.maxSinglePayloadBytes ?? MAX_SINGLE_PAYLOAD_BYTES
  let allocatedBytes = 0

  return {
    getAllocatedBytes(): number {
      return allocatedBytes
    },

    reserve(bytesOrObj: number | unknown): void {
      const bytesToReserve =
        typeof bytesOrObj === 'number'
          ? bytesOrObj
          : estimateObjectSizeBytes(bytesOrObj)

      if (bytesToReserve > maxSingleBytes) {
        const errorMsg = `SINGLE_PAYLOAD_EXCEEDS_CEILING: Payload size (${bytesToReserve} bytes) exceeds maximum single payload limit (${maxSingleBytes} bytes).`
        console.error(`>> [OOM:SINGLE_REJECT] ${errorMsg} [FAIL]`)
        throw new Error(`ISOLATE_MEMORY_BUDGET_EXCEEDED: ${errorMsg}`)
      }

      if (allocatedBytes + bytesToReserve > thresholdBytes) {
        const total = allocatedBytes + bytesToReserve
        const errorMsg = `TRIPWIRE_LIMIT_BREACHED: Allocation of ${bytesToReserve} bytes would push total memory to ${total} bytes (Threshold: ${thresholdBytes} bytes).`
        console.error(
          `>> [OOM:TRIPWIRE] ${errorMsg} Halting pipeline to prevent isolate termination [TRAPPED]`,
        )

        if (options.onTripwireExceeded) {
          try {
            options.onTripwireExceeded(total, thresholdBytes)
          } catch (cbErr: any) {
            console.error(
              `>> [OOM:HOOK_ERROR] Alert hook failed: ${cbErr.message}`,
            )
          }
        }

        throw new Error(`ISOLATE_MEMORY_BUDGET_EXCEEDED: ${errorMsg}`)
      }

      allocatedBytes += bytesToReserve
    },

    release(bytes: number): void {
      allocatedBytes = Math.max(0, allocatedBytes - Math.max(0, bytes))
    },

    reset(): void {
      allocatedBytes = 0
    },
  }
}
```

```typescript
import { describe, it, expect, vi } from 'vitest'
import { createIsolateContext } from '../../src/context/isolateContext.js'
import {
  createMemoryTripwire,
  estimateObjectSizeBytes,
  DEFAULT_OOM_TRIPWIRE_THRESHOLD_BYTES,
  MAX_SINGLE_PAYLOAD_BYTES,

import type { Env } from '../../src/tools.js'

describe('Isolate Memory Hygiene & Context Segregation Battery (Phase 3)', () => {
  const mockEnvA = {
    GITHUB_TOKEN: 'ghp_token_alpha',
    CLOUDFLARE_ACCOUNT_ID: 'acc_alpha',
  } as unknown as Env

  const mockEnvB = {
    GITHUB_TOKEN: 'ghp_token_beta',
    CLOUDFLARE_ACCOUNT_ID: 'acc_beta',
  } as unknown as Env

  it('TASK-3.1: maintains strict context isolation across concurrent instances without global bleeding', () => {
    const ctxA = createIsolateContext(mockEnvA, undefined, {
      source: 'ROUTE_A',
    })
    const ctxB = createIsolateContext(mockEnvB, undefined, {
      source: 'ROUTE_B',
    })

    expect(ctxA.traceId).not.toEqual(ctxB.traceId)
    expect(ctxA.env.GITHUB_TOKEN).toBe('ghp_token_alpha')
    expect(ctxB.env.GITHUB_TOKEN).toBe('ghp_token_beta')

    ctxA.log('AUTH', 'User alpha verified')
    expect(ctxA.getTelemetryEntries()).toHaveLength(1)
    expect(ctxB.getTelemetryEntries()).toHaveLength(0)

    ctxB.warn('RATE_LIMIT', 'API quota at 80%')
    expect(ctxA.getTelemetryEntries()).toHaveLength(1)
    expect(ctxB.getTelemetryEntries()).toHaveLength(1)
    expect(ctxA.getTelemetryEntries()[0].tag).toBe('AUTH')
    expect(ctxB.getTelemetryEntries()[0].tag).toBe('RATE_LIMIT')
  })

  it('TASK-3.1: guarantees globalThis is not polluted with ambient variables', () => {
    const initialGlobals = Object.keys(globalThis)
    createIsolateContext(mockEnvA)
    const postInitGlobals = Object.keys(globalThis)
    expect(postInitGlobals).toEqual(initialGlobals)
  })

  it('TASK-3.2: estimates object size accurately and handles circular references safely', () => {
    // Primitive sizing
    expect(estimateObjectSizeBytes(1234)).toBe(8)
    expect(estimateObjectSizeBytes('hello')).toBe(10) // 5 chars * 2 bytes
    expect(estimateObjectSizeBytes(true)).toBe(4)

    // TypedArray sizing
    const buffer = new Uint8Array(1024)
    expect(estimateObjectSizeBytes(buffer)).toBe(1024)

    // Cyclic structure detection
    const cyclicObj: any = { name: 'cyclic', nested: {} }
    cyclicObj.nested.ref = cyclicObj

    // Must calculate cleanly without blowing the call stack
    const calculatedSize = estimateObjectSizeBytes(cyclicObj)
    expect(calculatedSize).toBeGreaterThan(0)
  })

  it('TASK-3.2: rejects single payloads exceeding MAX_SINGLE_PAYLOAD_BYTES', () => {
    const tripwire = createMemoryTripwire()
    const oversizedBytes = MAX_SINGLE_PAYLOAD_BYTES + 1

    expect(() => {
      tripwire.reserve(oversizedBytes)
    }).toThrow(/SINGLE_PAYLOAD_EXCEEDS_CEILING/)
    expect(tripwire.getAllocatedBytes()).toBe(0)
  })

  it('TASK-3.2: trips fail-closed OOM event and invokes alert callback when threshold is breached', () => {
    const alertHook = vi.fn()
    const tripwire = createMemoryTripwire({
      tripwireThresholdBytes: 10_000,
      maxSinglePayloadBytes: 8_000,
      onTripwireExceeded: alertHook,
    })

    // Reserve 6,000 bytes (within budget)
    tripwire.reserve(6_000)
    expect(tripwire.getAllocatedBytes()).toBe(6_000)
    expect(alertHook).not.toHaveBeenCalled()

    // Attempting to reserve another 5,000 bytes pushes total to 11,000 > 10,000
    expect(() => {
      tripwire.reserve(5_000)
    }).toThrow(/ISOLATE_MEMORY_BUDGET_EXCEEDED/)

    expect(alertHook).toHaveBeenCalledTimes(1)
    expect(alertHook).toHaveBeenCalledWith(11_000, 10_000)
    expect(tripwire.getAllocatedBytes()).toBe(6_000) // Unaltered allocation state
  })

  it('TASK-3.2: accurately tracks incremental memory reservations and releases', () => {
    const tripwire = createMemoryTripwire({
      tripwireThresholdBytes: DEFAULT_OOM_TRIPWIRE_THRESHOLD_BYTES,
    })

    const chunkA = 1024 * 1024 * 10 // 10 MB
    const chunkB = 1024 * 1024 * 15 // 15 MB

    tripwire.reserve(chunkA)
    expect(tripwire.getAllocatedBytes()).toBe(chunkA)

    tripwire.reserve(chunkB)
    expect(tripwire.getAllocatedBytes()).toBe(chunkA + chunkB)

    tripwire.release(chunkA)
    expect(tripwire.getAllocatedBytes()).toBe(chunkB)

    tripwire.reset()
    expect(tripwire.getAllocatedBytes()).toBe(0)
  })
})
```
