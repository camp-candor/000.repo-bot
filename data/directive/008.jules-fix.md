const formatPayload = (rawText: string) => {
    const trimmed = rawText.trim();
    if (!trimmed) return '   [EMPTY RESPONSE BODY - ISOLATE IDLE]';
    try {
        const parsed = JSON.parse(trimmed);
        return JSON.stringify(parsed, null, 2)
            .split('\n')
            .map((line) => `   ${line}`)
            .join('\n');
    } catch {
        return trimmed
            .split('\n')
            .map((line) => `   ${line}`)
            .join('\n');
    }
};
