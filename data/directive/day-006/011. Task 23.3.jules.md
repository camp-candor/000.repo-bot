### II. ARCHITECTURAL MISSION & REQUIREMENTS

Instrument the edge subsystems in `apps/worker` with broadcaster hooks so that
core lifecycle events automatically push real-time, monotonic 7-bit ASCII
telemetry frames through the `RepoBotDO` WebSocket Hibernation Engine directly
into connected terminal viewports (`cns00`):

1. **Central Telemetry Dispatcher (`apps/worker/src/RepoBotDO.ts`):**

- Export `emitEdgeTelemetry(env, type, source, payload, asciiMsg)`.

- Query the global Durable Object singleton
  (`env.REPO_BOT_DO.idFromName('global')`).

- Execute either direct Workers RPC (`stub.broadcastTelemetry(...)`) or forward
  an internal HTTP request (`POST https://internal/broadcast`) to ensure
  seamless execution in both unit tests and Cloudflare edge isolates.

- Mount `POST /broadcast` within `RepoBotDO.fetch` to ingest internal subsystem
  dispatches.

2. **Hot D1 Ledger Appends (`apps/worker/src/audit/auditLedger.ts`):**

- Accept optional `env` in `appendAuditEvent(db, event, env?)`.

- After inserting an event into the D1 `audit_events` table and generating its
  SHA-256 hash, call `emitEdgeTelemetry` with `type: 'AUDIT_LOG'` and formatted
  ASCII string:
  `>> [AUDIT #${nextSeq}] ${event.eventType} on ${event.repository} (Hash: ${recordHash.slice(0, 8)})`.

3. **PR Invariant & Scope Check Verification
   (`apps/worker/src/prAuditEngine.ts`):**

- At the completion of
  `auditPullRequest(owner, repo, pullNumber, headSha, baseSha, prBody, env, branchName)`,
  broadcast `type: 'PR_EVENT'`:

- If passed:
  `>> [SCOPE CHECK OK] PR #${pullNumber}@${headSha.slice(0, 7)}: ${files.length} file(s) whitelisted.`

- If failed:
  `>> [SCOPE VIOLATION] PR #${pullNumber}@${headSha.slice(0, 7)}: ${violations[0]}`

4. **Jules Agent Lifecycle Events (`apps/worker/src/jules.ts`):**

- In `dispatchJulesJob`, broadcast `type: 'JULES_EVENT'` upon cutting the
  ephemeral branch and spinning up the session:
  `>> [JULES DISPATCH] Task ${taskId} dispatched (Session: ${sessionId.slice(0, 10)}) on ${branchName}`.

- In `pollActiveJulesSessions`:
- When status is `AWAITING_USER_INPUT`, `NEEDS_ATTENTION`, or `PAUSED`:
  `>> [JULES] Session ${item.session_id.slice(0, 10)} requires operator input in #ask-jules`.

- When status reaches terminal `COMPLETED` or `FAILED`:
  `>> [JULES] Session ${item.session_id.slice(0, 10)} transitioned to ${currentState}`.

5. **Inbound Webhook Events (`apps/worker/src/index.ts`):**

- On workstation push events (`githubEvent === 'push'`), broadcast:
  `>> [PUSH] ${pusher} pushed ${headCommitSha.slice(0, 7)} to ${repo}@${branch}`.

- On PR merge completion (`pull_request.closed` & `merged === true`), broadcast:
  `>> [MERGE] PR #${pullNumber} merged into ${baseRef} (${taskId}) by ${mergedBy} [OK]`.

- Pass `c.env` into `appendAuditEvent` on pull request webhooks.

6. **Dedicated Test Suite
   (`apps/worker/test/unit/telemetryBroadcasterHooks.test.ts`):**

- Verify that each hook formats its payload correctly and dispatches via
  `emitEdgeTelemetry`.

- Assert non-blocking resilience: ensure failing or missing DO bindings do not
  throw unhandled exceptions.

/**
 * Utility for edge subsystems to broadcast telemetry to connected operators.
 * Non-blocking, fails safe if REPO_BOT_DO is unconfigured or unreachable.
 */
export async function emitEdgeTelemetry(
  env: any,
  type: TelemetryPacket['type'],
  source: string,
  payload: any,
  asciiMsg: string,
): Promise<void> {
  if (!env?.REPO_BOT_DO) return
  try {
    const id = env.REPO_BOT_DO.idFromName('global')
    const stub = env.REPO_BOT_DO.get(id)
    if (typeof stub.broadcastTelemetry === 'function') {
      await stub.broadcastTelemetry(type, source, payload, asciiMsg)
    } else {
      await stub.fetch(
        new Request('https://internal/broadcast', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            type,
            source,
            payload,
            ascii: asciiMsg,
          }),
        }),
      )
    }
  } catch (err: any) {
    console.warn('[TELEMETRY_EMIT_WARN]', err.message)
  }
}
```

import { computeGenesisHash, computeRecordHash } from './hashChain.js'
import { emitEdgeTelemetry } from '../RepoBotDO.js'

export interface AuditEventInput {
  taskId: string
  repository: string
  eventType: string
  actorId: string
  headSha: string
  payload: any
}

export interface StoredAuditEvent {
  sequence_id: number
  task_id: string
  repository: string
  event_type: string
  actor_id: string
  head_sha: string
  payload_json: string
  prev_hash: string
  record_hash: string
  created_at: number
  drained_at: number | null
}

const INIT_SCHEMA_SQL = `
CREATE TABLE IF NOT EXISTS audit_events (
    sequence_id INTEGER PRIMARY KEY AUTOINCREMENT,
    task_id TEXT NOT NULL,
    repository TEXT NOT NULL,
    event_type TEXT NOT NULL,
    actor_id TEXT NOT NULL,
    head_sha TEXT NOT NULL,
    payload_json TEXT NOT NULL,
    prev_hash TEXT NOT NULL,
    record_hash TEXT NOT NULL,
    created_at INTEGER NOT NULL,
    drained_at INTEGER DEFAULT NULL
);
CREATE INDEX IF NOT EXISTS idx_audit_task ON audit_events (task_id, sequence_id);
CREATE INDEX IF NOT EXISTS idx_audit_undrained ON audit_events (drained_at) WHERE drained_at IS NULL;
CREATE INDEX IF NOT EXISTS idx_audit_repo_seq ON audit_events (repository, sequence_id DESC);
`

let isSchemaInitialized = false

export async function ensureAuditSchema(db: any): Promise<void> {
  if (isSchemaInitialized || !db) return
  try {
    if (typeof db.exec === 'function') {
      await db.exec(INIT_SCHEMA_SQL)
      isSchemaInitialized = true
    }
  } catch (err) {
    console.error(
      '[D1_BOOTSTRAP_ERROR] Failed to auto-initialize audit schema:',
      err,
    )
  }
}

/**
 * Appends an event to the D1 Hot Transactional Ledger with SHA-256 chaining
 * and non-blocking real-time telemetry broadcasting.
 */
export async function appendAuditEvent(
  db: any,
  event: AuditEventInput,
  env?: any,
): Promise<StoredAuditEvent> {
  await ensureAuditSchema(db)
  const now = Date.now()

  // 1. Fetch latest record for this repository to retrieve Hn-1
  const lastRow = await db
    .prepare(
      'SELECT sequence_id, record_hash FROM audit_events WHERE repository = ? ORDER BY sequence_id DESC LIMIT 1',
    )
    .bind(event.repository)
    .first()

  let prevHash = lastRow?.record_hash
  if (!prevHash) {
    prevHash = await computeGenesisHash(event.repository)
  }

  // 2. Determine sequence ID
  const maxSeqRow = await db
    .prepare('SELECT MAX(sequence_id) as maxSeq FROM audit_events')
    .first()
  const nextSeq = (maxSeqRow?.maxSeq || 0) + 1

  // 3. Compute Hn
  const { recordHash, canonicalPayload } = await computeRecordHash(prevHash, {
    sequenceId: nextSeq,
    taskId: event.taskId,
    repository: event.repository,
    eventType: event.eventType,
    actorId: event.actorId,
    headSha: event.headSha,
    payload: event.payload,
    createdAt: now,
  })

  // 4. Insert into D1
  await db
    .prepare(
      `INSERT INTO audit_events
            (sequence_id, task_id, repository, event_type, actor_id, head_sha, payload_json, prev_hash, record_hash, created_at)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    )
    .bind(
      nextSeq,
      event.taskId,
      event.repository,
      event.eventType,
      event.actorId,
      event.headSha,
      canonicalPayload,
      prevHash,
      recordHash,
      now,
    )
    .run()

  // 5. Broadcast live telemetry frame to connected operators
  if (env) {
    emitEdgeTelemetry(
      env,
      'AUDIT_LOG',
      'auditLedger',
      {
        seq: nextSeq,
        repo: event.repository,
        type: event.eventType,
        hash: recordHash.slice(0, 8),
      },
      `>> [AUDIT #${nextSeq}] ${event.eventType} on ${event.repository} (Hash: ${recordHash.slice(0, 8)})`,
    ).catch(() => {})
  }

  return {
    sequence_id: nextSeq,
    task_id: event.taskId,
    repository: event.repository,
    event_type: event.eventType,
    actor_id: event.actorId,
    head_sha: event.headSha,
    payload_json: canonicalPayload,
    prev_hash: prevHash,
    record_hash: recordHash,
    created_at: now,
    drained_at: null,
  }
}
```
