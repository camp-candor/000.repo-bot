### II. ARCHITECTURAL MISSION & REQUIREMENTS

Implement the edge-side **Historical Telemetry Replay Engine ($K=10$ Backlog
Buffer)** in `RepoBotDO.ts`:

1. **State Persistence & FIFO Ring Buffer (`TASK-23.8.1`):**

- Maintain a private in-memory cache:
  `private recentTelemetry: TelemetryPacket[] = []`.
- Implement cold-start hydration: When isolate memory is cold
  (`this.recentTelemetry.length === 0`), read key `'recent_telemetry'` from NVMe
  storage (`this.ctx.storage.get<TelemetryPacket[]>('recent_telemetry')`).
  Default to an empty array `[]` if uninitialized.
- In `broadcastTelemetry(type, source, payload, asciiMsg)`:
- Ignore internal batch envelopes (`type === 'TELEMETRY_HISTORY'`) to avoid
  recursive buffering.
- Append the stamped `TelemetryPacket` onto `this.recentTelemetry`.
- Clamp the ring buffer strictly to the latest 10 packets
  (`if (this.recentTelemetry.length > 10) this.recentTelemetry = this.recentTelemetry.slice(-10)`).
- Asynchronously persist the clamped array to transactional storage:
  `this.ctx.storage.put('recent_telemetry', this.recentTelemetry).catch(() => {})`.

2. **Wire Protocol Envelope & Handshake Ingress (`TASK-23.8.2`):**

- Update `TelemetryPacket['type']` union to include `'TELEMETRY_HISTORY'`.
- In `RepoBotDO.fetch` handling `/ws/telemetry` and `/ws` upgrade handshakes:

1. Verify `request.headers.get('Upgrade') === 'websocket'`; reject non-websocket
   requests with `HTTP 426 Upgrade Required`.
2. Accept the server socket descriptor via
   `this.ctx.acceptWebSocket(server, ['operator'])`.
3. Bind metadata:
   `server.serializeAttachment({ connectedAt: Date.now(), role: ... })`.
4. Emit the initial `HEARTBEAT` frame down the newly accepted socket.
5. Hydrate `this.recentTelemetry` from storage if the in-memory array is empty.
6. If `this.recentTelemetry.length > 0`, emit a single `TELEMETRY_HISTORY`
   envelope down the connecting socket (`server.send(...)`) containing the
   chronologically ordered array of up to 10 historical packets:

```typescript
const historyPacket: TelemetryPacket = {
  seq: this.telemetrySeq,
  ts: Date.now(),
  type: 'TELEMETRY_HISTORY',
  source: 'RepoBotDO',
  payload: {
    count: this.recentTelemetry.length,
    items: [...this.recentTelemetry],
  },
  ascii: `>> [TELEMETRY] Replaying last ${this.recentTelemetry.length} edge event(s)...`,
}
server.send(JSON.stringify(historyPacket))
```

7. Note: The historical replay envelope MUST ONLY be transmitted to the newly
   connecting socket (`server`), NEVER rebroadcast across existing operator
   sockets.
8. Return `new Response(null, { status: 101, webSocket: client })`.

9. **Dedicated Vitest Unit Battery
   (`apps/worker/test/unit/telemetryWebSocket.test.ts`):**

- Verify HTTP 426 rejection for non-WebSocket requests.
- Verify HTTP 101 upgrade handshake and initial `HEARTBEAT` emission.
- Verify that connecting to `/ws/telemetry` with 0 historical items emits zero

- Verify that connecting with seeded items in `recent_telemetry` emits a single

- Verify that calling `broadcastTelemetry` more than 10 times clamps stored

- Verify that the historical replay is only sent to the connecting socket, not

```typescript
import { DurableObject } from 'cloudflare:workers'
import type { Env } from './tools.js'

export interface TelemetryPacket {
  seq: number
  ts: number
  type:
    | 'HEARTBEAT'
    | 'TASK_TRANSITION'
    | 'AUDIT_LOG'
    | 'PR_EVENT'
    | 'JULES_EVENT'
    | 'SLACK_RECEIPT'
    | 'TELEMETRY_HISTORY'
  source: string
  payload: Record<string, any>
  ascii: string
}

export interface TelemetryHistoryPacket {
  seq: number
  ts: number
  type: 'TELEMETRY_HISTORY'
  source: 'RepoBotDO'
  payload: {
    count: number
    items: TelemetryPacket[]
  }
  ascii: string
}

/**
 * Utility for edge subsystems to broadcast telemetry to connected operators.
 * Non-blocking, fails safe if REPO_BOT_DO is unconfigured or unreachable.
 */
export async function emitEdgeTelemetry(
  env: any,
  type: TelemetryPacket['type'],
  source: string,
  payload: any,
  asciiMsg: string,
): Promise<void> {
  if (!env?.REPO_BOT_DO) return
  try {
    const id = env.REPO_BOT_DO.idFromName('global')
    const stub = env.REPO_BOT_DO.get(id)
    if (typeof stub.broadcastTelemetry === 'function') {
      await stub.broadcastTelemetry(type, source, payload, asciiMsg)
    } else {
      await stub.fetch(
        new Request('https://internal/broadcast', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            type,
            source,
            payload,
            ascii: asciiMsg,
          }),
        }),
      )
    }
  } catch (err: any) {
    console.warn('[TELEMETRY_EMIT_WARN]', err.message)
  }
}

export class RepoBotDO extends DurableObject<Env> {
  private telemetrySeq = 0
  private recentTelemetry: TelemetryPacket[] = []

  constructor(ctx: DurableObjectState, env: Env) {
    super(ctx, env)
  }

  /**
   * Broadcasts a telemetry packet to all connected operator WebSockets
   * and maintains the rolling NVMe FIFO buffer (K=10).
   */
  async broadcastTelemetry(
    type: TelemetryPacket['type'],
    source: string,
    payload: any,
    asciiMsg: string,
  ): Promise<void> {
    const packet: TelemetryPacket = {
      seq: ++this.telemetrySeq,
      ts: Date.now(),
      type,
      source,
      payload: payload || {},
      ascii: asciiMsg,
    }

    // Maintain rolling FIFO ring buffer (K=10)
    if (type !== 'TELEMETRY_HISTORY') {
      this.recentTelemetry.push(packet)
      if (this.recentTelemetry.length > 10) {
        this.recentTelemetry = this.recentTelemetry.slice(-10)
      }
      this.ctx.storage
        .put('recent_telemetry', this.recentTelemetry)
        .catch(() => {})
    }

    const sockets = this.ctx.getWebSockets('operator')
    if (sockets.length === 0) return

    const raw = JSON.stringify(packet)
    for (const ws of sockets) {
      try {
        ws.send(raw)
      } catch {
        try {
          ws.close(1011, 'Broadcast transmission failure')
        } catch {}
      }
    }
  }

  async webSocketMessage(
    ws: WebSocket,
    message: string | ArrayBuffer,
  ): Promise<void> {
    try {
      const str =
        typeof message === 'string'
          ? message
          : new TextDecoder().decode(message)
      if (str === 'PING' || str.includes('"type":"PING"')) {
        ws.send(JSON.stringify({ type: 'PONG', ts: Date.now() }))
      }
    } catch {}
  }

  async webSocketClose(
    ws: WebSocket,
    code: number,
    reason: string,
    wasClean: boolean,
  ): Promise<void> {
    try {
      ws.close(code, reason)
    } catch {}
  }

  async webSocketError(ws: WebSocket, error: unknown): Promise<void> {
    try {
      ws.close(1011, 'Unexpected socket error')
    } catch {}
  }

  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url)
    const path = url.pathname

    // 1. WebSocket Ingress Handshake (/ws/telemetry and /ws)
    if (path === '/ws/telemetry' || path === '/ws') {
      if (request.headers.get('Upgrade') !== 'websocket') {
        return new Response('Expected Upgrade: websocket', { status: 426 })
      }

      const pair = new WebSocketPair()
      const [client, server] = Object.values(pair)

      this.ctx.acceptWebSocket(server, ['operator'])
      server.serializeAttachment({
        connectedAt: Date.now(),
        role: url.searchParams.get('role') || 'operator',
      })

      // 1.1 Emit Initial Edge Handshake Frame
      const heartbeatPacket: TelemetryPacket = {
        seq: ++this.telemetrySeq,
        ts: Date.now(),
        type: 'HEARTBEAT',
        source: 'RepoBotDO',
        payload: {
          status: 'ONLINE',
          role: url.searchParams.get('role') || 'operator',
        },
        ascii:
          '>> [TELEMETRY] Edge tunnel established. Live streaming engaged.',
      }
      server.send(JSON.stringify(heartbeatPacket))

      // 1.2 Hydrate and Replay Last 10 Telemetry Events (K=10)
      if (this.recentTelemetry.length === 0) {
        this.recentTelemetry =
          (await this.ctx.storage.get<TelemetryPacket[]>('recent_telemetry')) ||
          []
      }

      if (this.recentTelemetry.length > 0) {
        const historyPacket: TelemetryPacket = {
          seq: this.telemetrySeq,
          ts: Date.now(),
          type: 'TELEMETRY_HISTORY',
          source: 'RepoBotDO',
          payload: {
            count: this.recentTelemetry.length,
            items: [...this.recentTelemetry],
          },
          ascii: `>> [TELEMETRY] Replaying last ${this.recentTelemetry.length} edge event(s)...`,
        }
        server.send(JSON.stringify(historyPacket))
      }

      return new Response(null, { status: 101, webSocket: client })
    }

    // 2. Internal Subsystem Broadcast Ingress
    if (request.method === 'POST' && path === '/broadcast') {
      const body: any = await request.json().catch(() => null)
      if (body && body.type && body.ascii) {
        await this.broadcastTelemetry(
          body.type,
          body.source || 'edgeSubsystem',
          body.payload || {},
          body.ascii,
        )
      }
      return Response.json({ ok: true })
    }

    // 3. FSM Transition Ingress
    if (request.method === 'POST' && path === '/fsm/transition') {
      const body: any = await request.json().catch(() => null)
      const taskId = body?.taskId || 'TASK-UNKNOWN'
      const targetState = body?.state || 'IDLE'

      await this.broadcastTelemetry(
        'TASK_TRANSITION',
        'RepoBotDO',
        { taskId, state: targetState },
        `>> [FSM] Task ${taskId} transitioned to ${targetState}`,
      )
      return Response.json({ ok: true, taskId, state: targetState })
    }

    return new Response('Not Found', { status: 404 })
  }
}
```
